// ---- vue microscopique : ions carboxylate, micelle autour d'une goutte de graisse, ions Ca²⁺
function openMicelle(mode) {
  openMicro({
    title: 'Vue microscopique · ions carboxylate et goutte de graisse',
    caption: mode === 'ca' ? `En présence d'ions calcium Ca²⁺ (verts), les têtes –CO₂⁻ s'associent deux par deux aux ions Ca²⁺ : il se forme un solide (le trouble) et il reste moins d'ions carboxylate pour former des micelles.`
      : `Chaque ion carboxylate a une <b>tête hydrophile</b> –CO₂⁻ (rouge), tournée vers l'eau, et une <b>chaîne lipophile</b> (gris), plongée dans la goutte de graisse (jaune). Les ions s'organisent en <b>micelle</b> : la graisse est emportée par l'eau de rinçage.`,
    sim: sc => {
      const grease = new THREE.Mesh(new THREE.SphereGeometry(1.1, 32, 24), new THREE.MeshStandardMaterial({ color: 0xffd54f, transparent: true, opacity: .55, roughness: .3 })); grease.position.y = 2; sc.add(grease);
      const lab = label('goutte de graisse', { size: 36, scale: .006 }); lab.position.set(0, 3.5, 0); sc.add(lab);
      const mol = () => { const g = new THREE.Group(); const cm = new THREE.MeshStandardMaterial({ color: 0x9e9e9e }); for (let i = 0; i < 7; i++) { const s = new THREE.Mesh(new THREE.SphereGeometry(.09, 10, 8), cm); s.position.y = -i * .15; g.add(s); }
        const hd = new THREE.Mesh(new THREE.SphereGeometry(.16, 14, 10), new THREE.MeshStandardMaterial({ color: 0xe53935, emissive: 0x501010 })); hd.position.y = .15; g.add(hd); return g; };
      const items = [], N = mode === 'ca' ? 14 : 34;
      for (let i = 0; i < N; i++) { const m = mol(); const dir = new THREE.Vector3(rnd(-1, 1), rnd(-1, 1), rnd(-1, 1)).normalize(); m.userData = { dir, t: rnd(0, 6) }; sc.add(m); items.push(m); }
      // ions libres et ions calcium
      const free = []; for (let i = 0; i < (mode === 'ca' ? 10 : 6); i++) { const m = mol(); m.position.set(rnd(-3.5, 3.5), rnd(.5, 4), rnd(-1.5, 1.5)); m.rotation.set(rnd(0, 6), rnd(0, 6), 0); sc.add(m); free.push(m); }
      const cas = []; if (mode === 'ca') for (let i = 0; i < 5; i++) { const c = new THREE.Mesh(new THREE.SphereGeometry(.2, 14, 10), new THREE.MeshStandardMaterial({ color: 0x43a047, emissive: 0x103010 })); c.add(Object.assign(ionLabel('Ca²⁺'), {})); c.position.copy(free[2 * i].position).add(new THREE.Vector3(.25, .2, 0)); sc.add(c); cas.push(c); }
      if (mode === 'ca') { const pl = label('association Ca²⁺ / carboxylate → solide', { size: 32, scale: .006, color: '#fff', bg: 'rgba(11,20,48,.7)' }); pl.position.set(0, -.6, 1.5); sc.add(pl); }
      let T = 0;
      return dt => {
        T += dt; grease.rotation.y += dt * .2;
        items.forEach((m, i) => { const d = m.userData.dir.clone().applyAxisAngle(new THREE.Vector3(0, 1, 0), T * .2); m.position.copy(grease.position).addScaledVector(d, 1.25); m.quaternion.setFromUnitVectors(new THREE.Vector3(0, 1, 0), d); });
        free.forEach((m, i) => { m.position.y += Math.sin(T + i) * dt * .1; m.rotation.z += dt * .3; if (mode === 'ca') m.position.y -= dt * .05; });
        cas.forEach((c, i) => { c.position.y -= dt * .05; });
        if (mode === 'ca' && T > 20) { T = 0; free.forEach(m => m.position.y += 1); cas.forEach(c => c.position.y += 1); }
      };
    },
  });
}
