function Graphe(canvas) {
  const ctx = canvas.getContext("2d");
  function pas(etendue) { const b = 10 ** Math.floor(Math.log10(etendue / 5)), m = etendue / 5 / b; return b * (m < 1.5 ? 1 : m < 3.5 ? 2 : m < 7.5 ? 5 : 10); }
  function trace(cfg) {
    const W = canvas.width, H = canvas.height, ink = css("--ink"), mut = css("--muted"), line = css("--line");
    ctx.clearRect(0, 0, W, H); ctx.fillStyle = css("--surface"); ctx.fillRect(0, 0, W, H);
    const m = { g: 74, d: 18, h: 34, b: 52 }, pw = W - m.g - m.d, ph = H - m.h - m.b;
    let { xmin, xmax, ymin, ymax } = cfg;
    const tous = cfg.series.flatMap(s => s.pts || []);
    if (xmin == null) xmin = Math.min(0, ...tous.map(p => p[0])); if (xmax == null) xmax = Math.max(...tous.map(p => p[0]), xmin + 1e-9);
    if (ymin == null) ymin = Math.min(0, ...tous.map(p => p[1])); if (ymax == null) ymax = Math.max(...tous.map(p => p[1]), ymin + 1e-9);
    const py0 = (ymax - ymin) * 0.06; ymax += py0; if (ymin < 0) ymin -= py0;
    const X = x => m.g + (x - xmin) / (xmax - xmin) * pw, Y = y => m.h + ph - (y - ymin) / (ymax - ymin) * ph;
    ctx.font = "20px " + css("--font-body"); ctx.textBaseline = "middle";
    const sx = pas(xmax - xmin), sy = pas(ymax - ymin);
    ctx.strokeStyle = line; ctx.lineWidth = 1; ctx.fillStyle = mut;
    for (let v = Math.ceil(xmin / sx) * sx; v <= xmax + 1e-9; v += sx) { ctx.beginPath(); ctx.moveTo(X(v), m.h); ctx.lineTo(X(v), m.h + ph); ctx.stroke(); ctx.textAlign = "center"; ctx.fillText(fx(v, sx < 1 ? (sx < .1 ? 2 : 1) : 0), X(v), m.h + ph + 18); }
    for (let v = Math.ceil(ymin / sy) * sy; v <= ymax + 1e-9; v += sy) { ctx.beginPath(); ctx.moveTo(m.g, Y(v)); ctx.lineTo(m.g + pw, Y(v)); ctx.stroke(); ctx.textAlign = "right"; ctx.fillText(fx(v, sy < 1 ? (sy < .1 ? 2 : 1) : 0), m.g - 8, Y(v)); }
    ctx.strokeStyle = ink; ctx.lineWidth = 1.6; ctx.beginPath(); ctx.moveTo(m.g, Y(Math.max(ymin, Math.min(ymax, 0)))); ctx.lineTo(m.g + pw, Y(Math.max(ymin, Math.min(ymax, 0)))); ctx.moveTo(X(Math.max(xmin, 0)), m.h); ctx.lineTo(X(Math.max(xmin, 0)), m.h + ph); ctx.stroke();
    ctx.fillStyle = ink; ctx.font = "700 20px " + css("--font-body"); ctx.textAlign = "right"; ctx.fillText(cfg.xlab || "", m.g + pw, m.h + ph + 40); ctx.textAlign = "left"; ctx.fillText(cfg.ylab || "", 8, 16);
    ctx.save(); ctx.beginPath(); ctx.rect(m.g, m.h - 4, pw, ph + 8); ctx.clip();
    cfg.series.forEach(s => {
      ctx.strokeStyle = ctx.fillStyle = s.c || ink; ctx.lineWidth = s.w || 2.5; ctx.setLineDash(s.dash || []);
      if (s.f) { ctx.beginPath(); for (let i = 0; i <= 300; i++) { const x = xmin + (xmax - xmin) * i / 300, y = s.f(x); i ? ctx.lineTo(X(x), Y(y)) : ctx.moveTo(X(x), Y(y)); } ctx.stroke(); }
      else if (s.points) s.pts.forEach(([x, y]) => { ctx.beginPath(); if (s.croix) { ctx.lineWidth = 2; ctx.moveTo(X(x) - 5, Y(y) - 5); ctx.lineTo(X(x) + 5, Y(y) + 5); ctx.moveTo(X(x) - 5, Y(y) + 5); ctx.lineTo(X(x) + 5, Y(y) - 5); ctx.stroke(); } else { ctx.arc(X(x), Y(y), s.r || 4, 0, 7); ctx.fill(); } });
      else { ctx.beginPath(); s.pts.forEach(([x, y], i) => i ? ctx.lineTo(X(x), Y(y)) : ctx.moveTo(X(x), Y(y))); ctx.stroke(); }
      ctx.setLineDash([]);
    });
    if (cfg.curseur != null) { ctx.strokeStyle = css("--amber"); ctx.lineWidth = 2; ctx.setLineDash([6, 5]); ctx.beginPath(); ctx.moveTo(X(cfg.curseur), m.h); ctx.lineTo(X(cfg.curseur), m.h + ph); ctx.stroke(); ctx.setLineDash([]); }
    ctx.restore();
    /* légende */
    ctx.font = "18px " + css("--font-body"); ctx.textAlign = "left";
    const noms = cfg.series.filter(s => s.nom); let lx = W - m.d - noms.reduce((a, s) => a + ctx.measureText(s.nom).width + 44, 0) + 20;

  }

}
/* régression polynomiale (moindres carrés), degré 1 ou 2, avec ou sans constante */
function ajuste(pts, deg, constante = true) {
  const base = x => { const b = []; if (constante) b.push(1); b.push(x); if (deg === 2) b.push(x * x); return b; };
  const n = base(0).length, A = Array.from({ length: n }, () => Array(n + 1).fill(0));
  pts.forEach(([x, y]) => { const b = base(x); for (let i = 0; i < n; i++) { for (let j = 0; j < n; j++) A[i][j] += b[i] * b[j]; A[i][n] += b[i] * y; } });
  for (let i = 0; i < n; i++) { let p = i; for (let k = i + 1; k < n; k++) if (Math.abs(A[k][i]) > Math.abs(A[p][i])) p = k; [A[i], A[p]] = [A[p], A[i]];
    for (let k = 0; k < n; k++) if (k !== i) { const f = A[k][i] / A[i][i]; for (let j = i; j <= n; j++) A[k][j] -= f * A[i][j]; } }
  const c = A.map((r, i) => r[n] / r[i]); const res = { c0: constante ? c[0] : 0, c1: c[constante ? 1 : 0], c2: deg === 2 ? c[constante ? 2 : 1] : 0 };
  res.f = x => res.c0 + res.c1 * x + res.c2 * x * x; return res;
}

/* ---------------------------------------------------------------------
   LABO 3D v3 : matériel de laboratoire réaliste (même mode d'emploi que la
   v1 et la v2 : mêmes fonctions, mêmes paramètres, mêmes tailles et positions).
   v3 : paillasse en résine (sans quadrillage), burette gravée tous les 0,1 mL
   avec ménisque, afficheurs LCD à 7 segments, boîtiers d'appareils clairs à
   clavier à membrane, compartiment à cuve du spectrophotomètre avec capot.
   Verrerie tournée (profils réels : bécher à bec, erlenmeyer, éprouvette sur
   pied, tube à essai), burette à robinet et bande de Schellbach, potence à
   socle lourd, appareils reconnus d'après leur nom, bidon et évier.
   --------------------------------------------------------------------- */
function Labo3D(V3) {
  const T = THREE, sc = V3.scene;
  const mat = (c, o = {}) => new T.MeshStandardMaterial(Object.assign({ color: c, roughness: .6 }, o));
  const verre = new T.MeshPhysicalMaterial({ color: 0xdfefff, transparent: true, opacity: .28, roughness: .05, metalness: 0, side: T.DoubleSide, depthWrite: false });
  const verreBord = new T.MeshPhysicalMaterial({ color: 0xeaf4ff, transparent: true, opacity: .55, roughness: .05, metalness: 0, depthWrite: false });
  const blanc = new T.MeshBasicMaterial({ color: 0xffffff, transparent: true, opacity: .9, depthWrite: false });
  const metal = mat(0xc4ccd4, { metalness: .55, roughness: .3 });
  const caoutchouc = mat(0x2b2f36, { roughness: .85 });
  const place = (m, x, y, z) => { m.position.set(x, y, z); sc.add(m); return m; };
  const etiq = (txt, x, y, z, o = {}) => place(V3.etiquette(txt, Object.assign({ taille: .22 }, o)), x, y, z);
  /* objet de révolution à partir d'un profil [[rayon, hauteur], …] */
  const tour = (prof, seg = 40) => new T.LatheGeometry(prof.map(([r, y]) => new T.Vector2(Math.max(r, 1e-4), y)), seg);
  /* boîte à coins arrondis (boîtiers d'appareils, bidon) */
  function boiteRonde(w, h, p, r) {
    r = Math.min(r, w / 2 - 1e-3, h / 2 - 1e-3);
    const s = new T.Shape(), x = -w / 2, y = -h / 2;
    s.moveTo(x + r, y); s.lineTo(x + w - r, y); s.quadraticCurveTo(x + w, y, x + w, y + r); s.lineTo(x + w, y + h - r);
    s.quadraticCurveTo(x + w, y + h, x + w - r, y + h); s.lineTo(x + r, y + h); s.quadraticCurveTo(x, y + h, x, y + h - r);
    s.lineTo(x, y + r); s.quadraticCurveTo(x, y, x + r, y);
    const b = Math.min(.02, p * .2);
    const g = new T.ExtrudeGeometry(s, { depth: Math.max(p - 2 * b, 1e-3), bevelEnabled: true, bevelThickness: b, bevelSize: b, bevelSegments: 2, curveSegments: 6 });
    g.translate(0, 0, -p / 2 + b); return g;
  }
  /* petite texture de texte (nom imprimé sur un boîtier, étiquette de bidon) */
  function texte(lignes, o = {}) {
    const cv = document.createElement("canvas"); cv.width = o.w || 256; cv.height = o.h || 64; const g = cv.getContext("2d");
    if (o.fond) { g.fillStyle = o.fond; g.fillRect(0, 0, cv.width, cv.height); }
    if (o.bande) { g.fillStyle = o.bande; g.fillRect(0, 0, cv.width, cv.height * .22); }
    g.fillStyle = o.couleur || "#ffffff"; g.textAlign = "center"; g.textBaseline = "middle";
    const L = [].concat(lignes), y0 = o.bande ? cv.height * .22 : 0, hh = (cv.height - y0) / L.length;
    L.forEach((t, i) => { g.font = `700 ${Math.round(hh * .55)}px Atkinson Hyperlegible, Segoe UI, sans-serif`; g.fillText(t, cv.width / 2, y0 + hh * (i + .5), cv.width * .92); });
    const tx = new T.CanvasTexture(cv); tx.anisotropy = 4; return tx;
  }

  /* ---------- paillasse : plan de travail en résine, chant clair ---------- */
  function paillasse(l = 9, p = 4, cx = .4) {
    /* plan de travail en résine époxy : teinte gris anthracite chaud, fin moucheté, reflets doux */
    const cv = document.createElement("canvas"); cv.width = cv.height = 256; const c = cv.getContext("2d");
    c.fillStyle = "#4d4a55"; c.fillRect(0, 0, 256, 256);
    for (let i = 0; i < 2600; i++) { c.fillStyle = Math.random() < .5 ? "rgba(255,255,255,.05)" : "rgba(0,0,0,.07)"; c.fillRect(Math.random() * 256, Math.random() * 256, 1.5, 1.5); }
    const tx = new T.CanvasTexture(cv); tx.wrapS = tx.wrapT = T.RepeatWrapping; tx.repeat.set(l / 2, p / 2);
    const plan = place(new T.Mesh(new T.BoxGeometry(l, .2, p), mat(0xffffff, { roughness: .42, metalness: .05, map: tx })), cx, -.1, 0); plan.receiveShadow = true;
    /* chant arrondi clair à l'avant et caisson du meuble */
    const chant = place(new T.Mesh(new T.CylinderGeometry(.06, .06, l, 12), mat(0xd9d6de, { roughness: .45 })), cx, -.06, p / 2); chant.rotation.z = Math.PI / 2;
    place(new T.Mesh(new T.BoxGeometry(l - .2, .6, .05), mat(0xc9c3d3, { roughness: .7 })), cx, -.5, p / 2 - .15);
  }

  /* ---------- liquide (cylindre ou tronc de cône si le récipient s'évase) ---------- */
  function liquide(r, couleur, op = .55, seg = 32, rBas) {
    const matL = new T.MeshStandardMaterial({ color: couleur, transparent: true, opacity: op, roughness: .1 });
    if (rBas == null || Math.abs(rBas - r) < 1e-3) {
      const m = new T.Mesh(new T.CylinderGeometry(r, r, 1, seg), matL); sc.add(m);
      m.regler = (y0, h, c) => { m.visible = h > 1e-4; m.scale.y = Math.max(h, 1e-4); m.position.y = y0 + Math.max(h, 1e-4) / 2; if (c != null) m.material.color.set(c); };
      return m;
    }
    /* récipient conique (erlenmeyer) : la surface du liquide rétrécit quand le niveau monte */
    const m = new T.Mesh(new T.CylinderGeometry(r, rBas, 1, seg), matL); sc.add(m); let hTot = 1, dernier = -1;
    m.regler = (y0, h, c) => {
      m.visible = h > 1e-4; const hh = Math.max(h, 1e-4);
      if (Math.abs(hh - dernier) > 1e-4) { dernier = hh; const rh = rBas + (r - rBas) * Math.min(1, hh / (m.userData.hc || hTot));
        m.geometry.dispose(); m.geometry = new T.CylinderGeometry(rh, rBas, hh, seg); }
      m.position.y = y0 + hh / 2; if (c != null) m.material.color.set(c);
    };
    m.cone = hc => { m.userData.hc = hc; }; return m;
  }

  /* ---------- récipients : bécher, erlenmeyer, éprouvette, tube à essai ---------- */
  function recipient(r, h, x, y0, z, o = {}) {
    const g = new T.Group(); g.position.set(x, 0, z);
    const fondR = o.fondR || 1, ep = Math.max(.006, r * .035);
    let liq;
    if (o.rond) {
      /* tube à essai / tube à fond rond : col ourlé */
      const prof = [[0, 0]]; for (let k = 1; k <= 8; k++) { const a = k / 8 * Math.PI / 2; prof.push([r * Math.sin(a), r - r * Math.cos(a)]); }
      prof.push([r, h + r], [r + ep * 1.6, h + r + ep * .6], [r + ep * 1.2, h + r + ep * 1.8]);
      const v = new T.Mesh(tour(prof.map(([a, b]) => [a, b - r]), 32), verre); v.position.y = y0; g.add(v);
      liq = liquide(r * .94, o.couleur || 0xcfe6f7, o.op || .5);
    } else if (fondR > 1.3) {
      /* erlenmeyer : fond plat large, panse conique, col droit ourlé */
      const R = r * fondR, hc = h * .72;
      const v = new T.Mesh(tour([[0, .004], [R - ep * 3, 0], [R, ep * 3], [r * 1.05, hc], [r, hc + h * .06], [r, h], [r + ep * 1.5, h + ep], [r + ep, h + ep * 2.2]], 40), verre);
      v.position.y = y0; g.add(v);
      liq = liquide(r * .96, o.couleur || 0xcfe6f7, o.op || .5, 32, R * .95); liq.cone(hc);
      etiquetteBlanche(g, R * .78, y0 + hc * .32, R, hc);
    } else {
      const tube = h / r > 3.2;
      /* bécher (bec verseur) ou éprouvette graduée (pied hexagonal) */
      const prof = [[0, .004], [r * fondR - ep * 2, 0], [r * fondR, ep * 2], [r, h * .98], [r + ep * 1.4, h], [r + ep * .6, h + ep * 1.2]];
      const v = new T.Mesh(tour(prof, 40), verre); v.position.y = y0; g.add(v);
      const bord = new T.Mesh(new T.TorusGeometry(r + ep * .7, ep * .9, 6, 40), verreBord); bord.rotation.x = Math.PI / 2; bord.position.y = y0 + h; g.add(bord);
      const bec = new T.Mesh(new T.TorusGeometry(r * .16, ep * 1.1, 6, 12, Math.PI), verreBord);
      bec.position.set(-r - ep, y0 + h - ep, 0); bec.rotation.set(Math.PI / 2, 0, Math.PI / 2); bec.scale.set(1, 1.5, 1); g.add(bec);
      if (tube && y0 < .05 && r < .2) {
        const pied = new T.Mesh(new T.CylinderGeometry(r * 2.1, r * 2.3, Math.max(.03, r * .25), 6), verreBord); pied.position.y = y0 + .015; g.add(pied);
        v.position.y = y0 + Math.max(.03, r * .25);
      }
      /* graduations blanches (5 traits, ou 10 pour une éprouvette) */
      const n = tube ? 10 : 5;
      for (let k = 1; k <= n; k++) {
        const yk = y0 + h * .1 + (h * .72) * k / n, long = tube ? k % 5 === 0 : k % 2 === 0;
        const t = new T.Mesh(new T.BoxGeometry(Math.max(.012, r * (long ? .45 : .28)), Math.max(.004, h * .006), .002), blanc);
        t.position.set(-r * .18, yk, r * fondR * .02 + r + .001); g.add(t);
      }
      if (!tube && r >= .2) etiquetteBlanche(g, r * .55, y0 + h * .55, r, h * .3);
      liq = liquide(r * .96, o.couleur || 0xcfe6f7, o.op || .5, 32, fondR !== 1 ? r * fondR * .96 : undefined);
      if (fondR !== 1) liq.cone(h);
    }
    sc.add(g); liq.position.x = x; liq.position.z = z; liq.regler(y0, 0);
    return { g, liq, r, h, y0, x, z, niveau: hh => liq.regler(y0, Math.min(hh, h * .98)) };
  }
  /* zone blanche dépolie où l'on écrit au crayon (bécher, erlenmeyer) */
  function etiquetteBlanche(g, l, y, r, hz) {
    const e = new T.Mesh(new T.PlaneGeometry(l * .7, Math.min(hz * .35, l * .35)), new T.MeshBasicMaterial({ color: 0xffffff, transparent: true, opacity: .55, depthWrite: false }));
    e.position.set(r * .3, y, r * .96); e.rotation.y = .3; g.add(e);
  }

  /* ---------- potence : socle lourd peint, tige inox, noix ---------- */
  function potence(x, z, h = 7) {
    const s = place(new T.Mesh(boiteRonde(1.4, 1, .1, .08), mat(0x2b2f36, { roughness: .45, metalness: .3 })), x, .05, z); s.rotation.x = -Math.PI / 2; s.castShadow = true;
    place(new T.Mesh(new T.CylinderGeometry(.09, .1, .08, 20), metal), x, .13, z);
    [[-.55, -.38], [.55, -.38], [-.55, .38], [.55, .38]].forEach(([a, b]) => place(new T.Mesh(new T.CylinderGeometry(.05, .05, .02, 12), caoutchouc), x + a, .005, z + b));
    place(new T.Mesh(new T.CylinderGeometry(.045, .045, h, 16), metal), x, h / 2, z).castShadow = true;
    place(new T.Mesh(new T.SphereGeometry(.05, 12, 8), metal), x, h, z);
  }

  /* ---------- burette graduée : 0 mL en haut ; set(V) règle le niveau ---------- */
  function burette(o) {
    const B = Object.assign({ x: 0, z: 0, haut: 6.4, bas: 2.4, pointe: 1.75, r: .065, vmax: 25, couleur: 0xbfe0ff }, o);
    const yTop = B.haut + .5;
    const tube = new T.Mesh(new T.CylinderGeometry(B.r, B.r, yTop - B.bas, 24, 1, true), verre); place(tube, B.x, (yTop + B.bas) / 2, B.z);
    /* col évasé en haut (remplissage à l'entonnoir) */
    place(new T.Mesh(tour([[B.r, 0], [B.r * 1.25, .08], [B.r * 1.35, .12]], 24), verreBord), B.x, yTop, B.z);
    /* bande de Schellbach : fond blanc et ligne bleue derrière le tube (lecture du ménisque) */
    const dos = new T.Mesh(new T.PlaneGeometry(B.r * 1.3, B.haut - B.bas + .1), new T.MeshBasicMaterial({ color: 0xffffff, transparent: true, opacity: .7, depthWrite: false }));
    place(dos, B.x, (B.haut + B.bas) / 2, B.z - B.r * .7);
    place(new T.Mesh(new T.PlaneGeometry(B.r * .25, B.haut - B.bas + .1), new T.MeshBasicMaterial({ color: 0x1f5fd1 })), B.x, (B.haut + B.bas) / 2, B.z - B.r * .68);
    /* corps du robinet en verre + clé PTFE (tourne quand on ouvre) */
    const yR = B.bas - .12;
    place(new T.Mesh(new T.CylinderGeometry(B.r * .6, B.r * .6, .14, 16, 1, true), verre), B.x, B.bas - .06, B.z);
    const corps = place(new T.Mesh(new T.CylinderGeometry(.045, .045, .2, 16), verreBord), B.x, yR, B.z); corps.rotation.x = Math.PI / 2;
    const rob = new T.Group(); place(rob, B.x, yR, B.z);
    const axe = new T.Mesh(new T.CylinderGeometry(.03, .03, .3, 12), mat(0xf2f2f2, { roughness: .4 })); axe.rotation.x = Math.PI / 2; rob.add(axe);
    const cle = new T.Mesh(new T.BoxGeometry(.32, .06, .05), mat(0x1f6fd1, { roughness: .4 })); cle.position.z = .17; rob.add(cle);
    const ecrou = new T.Mesh(new T.CylinderGeometry(.04, .04, .04, 6), mat(0x1f6fd1, { roughness: .4 })); ecrou.rotation.x = Math.PI / 2; ecrou.position.z = -.16; rob.add(ecrou);
    place(new T.Mesh(new T.CylinderGeometry(B.r * .4, .01, B.bas - .2 - B.pointe, 16, 1, true), verre), B.x, (B.bas - .2 + B.pointe) / 2, B.z);
    const liq = liquide(B.r * .92, B.couleur, .6, 20); liq.position.x = B.x; liq.position.z = B.z;
    /* gravure : traits tous les 0,1 mL (0,05 mL au-delà de 25 mL : non), chiffres à chaque mL, inscriptions de la classe de précision */
    { const H = 4096, cv = document.createElement("canvas"); cv.width = 256; cv.height = H; const c = cv.getContext("2d");
      const marge = .07, yPx = k => (marge + (1 - 2 * marge) * k / B.vmax) * H, pas = B.vmax > 50 ? 1 : .1;
      c.fillStyle = "#15212d"; c.strokeStyle = "#15212d"; c.textBaseline = "middle";
      for (let v = 0; v <= B.vmax + 1e-6; v += pas) {
        const k = Math.round(v / pas), y = yPx(v), mL = Math.abs(v - Math.round(v)) < 1e-6, demi = Math.abs(v * 2 - Math.round(v * 2)) < 1e-6;
        c.lineWidth = mL ? 7 : 4; const L = mL ? 120 : demi ? 82 : 58; c.beginPath(); c.moveTo(40, y); c.lineTo(40 + L, y); c.stroke();
        if (mL) { c.font = "700 64px Arial"; c.textAlign = "left"; c.fillText(String(Math.round(v)), 168, y); }
      }
      c.font = "700 40px Arial"; c.textAlign = "center"; c.fillText(B.vmax + " mL", 128, yPx(0) - 170); c.fillText("± 0,03  20 °C", 128, yPx(0) - 120); c.fillText("AS  Ex", 128, yPx(0) - 72);
      const tx = new T.CanvasTexture(cv); tx.anisotropy = 8;
      const hT = (B.haut - B.bas) / (1 - 2 * marge), yC = B.haut - (B.haut - B.bas) / 2;
      const gr = new T.Mesh(new T.CylinderGeometry(B.r * 1.015, B.r * 1.015, hT, 24, 1, true, -.75, 1.5), new T.MeshBasicMaterial({ map: tx, transparent: true, depthWrite: false }));
      place(gr, B.x, yC, B.z); }
    for (let k = 0; k <= B.vmax; k += 5) place(V3.etiquette(String(k), { taille: .09, fond: false }), B.x + .15, B.haut - k / B.vmax * (B.haut - B.bas), B.z + B.r);
    /* ménisque : anneau sombre sur la surface du liquide (on lit le bas du ménisque) */
    const men = new T.Mesh(new T.TorusGeometry(B.r * .82, B.r * .1, 6, 24), new T.MeshBasicMaterial({ color: 0x2a3a55, transparent: true, opacity: .55, depthWrite: false })); men.rotation.x = Math.PI / 2; place(men, B.x, B.haut, B.z);
    const fond = new T.Mesh(new T.CircleGeometry(B.r * .9, 24), new T.MeshBasicMaterial({ color: 0x9fb7d9, transparent: true, opacity: .45, depthWrite: false, side: T.DoubleSide })); fond.rotation.x = -Math.PI / 2; place(fond, B.x, B.haut, B.z);
    if (B.label) etiq(B.label, B.x, B.haut + .75, B.z, { taille: .24 });
    if (B.pince !== false) {
      /* pince à burette : mâchoires gainées + bras vers l'arrière */
      const yP = 5.2;
      [-1, 1].forEach(sx => { place(new T.Mesh(new T.BoxGeometry(.05, .16, .12), caoutchouc), B.x + sx * (B.r + .03), yP, B.z); });
      place(new T.Mesh(new T.BoxGeometry(B.r * 2 + .16, .05, .05), metal), B.x, yP, B.z - .08);
      const bras = place(new T.Mesh(new T.CylinderGeometry(.025, .025, .5, 10), metal), B.x, yP, B.z - .33); bras.rotation.x = Math.PI / 2;
      place(new T.Mesh(new T.BoxGeometry(.1, .1, .1), mat(0x2b2f36, { metalness: .4, roughness: .4 })), B.x, yP, B.z - .6);
    }
