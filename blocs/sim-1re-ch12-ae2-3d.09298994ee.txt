function posAt(k, cut) { // position (x, z) à la date k·τ
  if (!cut || k <= CUT_AT) { const t = TH0 + OM * k * TAU; return new THREE.Vector3(RR * Math.cos(t), 0, RR * Math.sin(t)); }
  const p5 = posAt(CUT_AT, false), t5 = TH0 + OM * CUT_AT * TAU, v = new THREE.Vector3(-Math.sin(t5), 0, Math.cos(t5)).multiplyScalar(VV);
  return p5.clone().addScaledVector(v, (k - CUT_AT) * TAU);
}
function vAt(i, cut) { // vecteur vitesse en Mi : tangent, de valeur MiMi+1/τ
  const a = posAt(i, cut), b = posAt(i + 1, cut), val = a.distanceTo(b) / TAU;
  if (cut && i >= CUT_AT) return b.clone().sub(a).normalize().multiplyScalar(val);
  const t = TH0 + OM * i * TAU; return new THREE.Vector3(-Math.sin(t), 0, Math.cos(t)).multiplyScalar(val);
}
