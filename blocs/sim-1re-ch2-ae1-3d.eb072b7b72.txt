  if (!reacts(m, i)) return `${cap(MET[m].nom)} + ${ION[i].nom} : aucun changement visible.`;
  if (i === 'Cu' && m === 'Zn') return 'Zinc + ions cuivre (II) Cu²⁺ : la solution bleue se décolore, un dépôt rouge-brun (cuivre) se forme sur le zinc.';

  return 'Zinc + ions fer (II) Fe²⁺ : la solution vert pâle se décolore, un dépôt noir (fer) se forme sur le zinc.';
}

    if (a.t === 'prep') {

        const s0 = T.sample.userData.mat.color.clone(), s1 = new THREE.Color(ION[a.ion].dep);
        await tween(5, p => { T.liquidMat.color.lerpColors(c0, c1, p); T.liquidMat.opacity = lerp(o0, o1, p); T.sample.userData.mat.color.lerpColors(s0, s1, Math.min(1, p * 1.3)); T.sample.userData.mat.metalness = lerp(.6, .15, p); T.sample.userData.mat.roughness = lerp(.45, .95, p); });
      } else await wait(1800);
      observe(obsText(a.metal, a.ion));
      say(obsText(a.metal, a.ion));
      if (tubes.filter(Boolean).length === NSLOT && ROWS.every(([m, i]) => tubes.some(t => t && t.m === m && t.i === i))) step('melanges');
      return;
    }
    if (a.t === 'naoh') {
      const T = tubes[a.slot]; if (!T || T.naoh) return false;
      select(a.slot, true);
      if (!INSTANT) { const x = T.group.position.x; flyTo([x + .2, .75, 1.2], [x, .3, 0], .8); }
      await dropInto(T, 0xf5f5f5, 4); T.naoh = true;

      else { ST.bidon++; say('Tubes vidés dans le bidon de récupération. Paillasse rangée.'); step('vider'); }
      await tween(.8, p => tubes.forEach(T => T && T.setLevel(T.level * (1 - p))));
      this.resetScene(); flyTo(...this.views.ensemble);
      return;
    }
  },
  micro() {
    const T = this.selected();

      <span class="eq">${T.m}(s) + ${i.sym}(aq) → ${m.ion}(aq) + ${T.i}(s)</span>

function chooseMetal(m) { chosenMetal = m; AE1.refreshPanel(); say(`Métal choisi : ${MET[m].nom}.` + (chosenIon ? '' : ' Choisis maintenant une solution.')); }
function chooseIon(i) { chosenIon = i; AE1.refreshPanel(); say(`Solution choisie : ${ION[i].nom}.` + (chosenMetal ? '' : ' Choisis maintenant un métal.')); }
function select(k, silent) {
  sel = k; tubes.forEach((T, j) => T && highlight(T, j === k)); AE1.refreshPanel();
  if (!silent && tubes[k]) { const T = tubes[k]; say(`Tube ${k + 1} sélectionné : ${MET[T.m].nom} + ${ION[T.i].nom}.`); }
}
function prepare() {
  if (!chosenMetal || !chosenIon) return say('Choisis un métal <b>et</b> une solution.');
  if (chosenMetal === chosenIon) return say(`Le ${MET[chosenMetal].nom} avec ses propres ions ne t'apprendra rien : choisis les ions <b>d'un autre métal</b>.`);
  if (tubes.some(t => t && t.m === chosenMetal && t.i === chosenIon)) return say('Ce mélange est déjà dans un tube.');
  const slot = tubes.findIndex(t => !t); if (slot < 0) return say('Le portoir est plein : vide les tubes dans le bidon pour recommencer.');
  act({ t: 'prep', slot, metal: chosenMetal, ion: chosenIon });
}
