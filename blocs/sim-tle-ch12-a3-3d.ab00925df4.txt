    <p class="pb"><span>Problématique :</span> quel est le principe de fonctionnement d'un accélérateur linéaire de particules chargées ?</p>
    <ol class="parcours"><li><b>Prédire</b> avant de lancer</li><li><b>Expérimenter</b> en 3D</li><li><b>Calculer</b> F et v<sub>B</sub></li><li><b>Confronter</b> avec Python</li><li><b>Garder une trace</b> (compte rendu)</li></ol>
    <div class="modele"><h3>Le dispositif et le modèle</h3><ul>
      <li><b>1 étage</b> = le condensateur plan du complément scientifique : la particule entre en A (plaque P<sub>A</sub>) sans vitesse et sort en B (plaque P<sub>B</sub>). Le champ E⃗ y est uniforme, de valeur E = |U<sub>AB</sub>| / d.</li>
      <li><b>Plusieurs étages</b> : des tubes creux en cuivre, alternativement chargés + et −. À l'intérieur d'un tube, il n'y a pas de champ : la vitesse reste constante. Le champ n'existe que dans les intervalles entre deux tubes.</li>
      <li>Seule force : F⃗ = q E⃗ (poids négligé, environ 10<sup>13</sup> fois plus petit). Mécanique classique. La scène n'est pas à l'échelle (d = 4,0 m dans le manuel) ; les valeurs affichées sont physiques.</li></ul></div>
    <details class="panel mots"><summary>📖 Mots utiles et données</summary><dl>
      <dt>Champ entre deux plaques</dt><dd>Direction perpendiculaire aux plaques ; sens de la plaque chargée + vers la plaque chargée − (sens des potentiels décroissants) ; valeur E = |U<sub>AB</sub>| / d (V·m<sup>−1</sup>).</dd>
      <dt>Force électrique</dt><dd>F⃗ = q E⃗ ; pour un électron, q = −e : F⃗ est opposée à E⃗.</dd>
      <dt>Travail de la force électrique</dt><dd>W<sub>A→B</sub>(F⃗) = q × U<sub>AB</sub>.</dd>
      <dt>Théorème de l'énergie cinétique</dt><dd>E<sub>c</sub>(B) − E<sub>c</sub>(A) = W<sub>A→B</sub>(F⃗).</dd>
      <dt>Données</dt><dd>m<sub>e</sub> = 9,1 × 10<sup>−31</sup> kg ; q = −e = −1,60 × 10<sup>−19</sup> C ; m<sub>p</sub> = 1,67 × 10<sup>−27</sup> kg ; g = 9,81 m·s<sup>−2</sup>.</dd>
    </dl></details>
    <details class="panel mots" id="aide"><summary>❓ Aide</summary><dl>

      <dt>Inversion de la tension</dt><dd>Cochée : la tension alternative change de signe pendant que la particule est dans un tube (les signes + et − des tubes s'échangent). Décochée : les charges des tubes restent fixes.</dd>
      <dt>Écrire un nombre</dt><dd>1,9 ou 1.9 (sans la puissance de 10, déjà écrite).</dd>

/* nombre à la française : n chiffres significatifs, virgule décimale */
function fr(x, n = 2) {
  if (!isFinite(x)) return "–";
  if (x === 0) return "0";
  const ex = Math.floor(Math.log10(Math.abs(x)));
  if (ex >= -2 && ex <= 3) { const d = Math.max(0, n - 1 - ex); return x.toFixed(d).replace(".", ","); }
  let m = x / 10 ** ex, ms = m.toFixed(n - 1);
  if (Math.abs(+ms) >= 10) return fr(x * 1.0000001, n);
  return `${ms.replace(".", ",")} × 10<sup>${ex < 0 ? "−" + -ex : ex}</sup>`;
}

/* ---------------------------------------------------------------------
   GRAPHIQUE 2D (canvas) : axes gradués, séries de points et courbes
   --------------------------------------------------------------------- */
function Graphe(canvas) {
  const ctx = canvas.getContext("2d");
  function pas(etendue) { const b = 10 ** Math.floor(Math.log10(etendue / 5)), m = etendue / 5 / b; return b * (m < 1.5 ? 1 : m < 3.5 ? 2 : m < 7.5 ? 5 : 10); }
  function trace(cfg) {
    const W = canvas.width, H = canvas.height, ink = css("--ink"), mut = css("--muted"), line = css("--line");
    ctx.clearRect(0, 0, W, H); ctx.fillStyle = css("--surface"); ctx.fillRect(0, 0, W, H);
    const m = { g: 74, d: 18, h: 34, b: 52 }, pw = W - m.g - m.d, ph = H - m.h - m.b;
    let { xmin, xmax, ymin, ymax } = cfg;
    const tous = cfg.series.flatMap(s => s.pts || []);
    if (xmin == null) xmin = Math.min(0, ...tous.map(p => p[0])); if (xmax == null) xmax = Math.max(...tous.map(p => p[0]), xmin + 1e-9);
    if (ymin == null) ymin = Math.min(0, ...tous.map(p => p[1])); if (ymax == null) ymax = Math.max(...tous.map(p => p[1]), ymin + 1e-9);
    const py0 = (ymax - ymin) * 0.06; ymax += py0; if (ymin < 0) ymin -= py0;
    const X = x => m.g + (x - xmin) / (xmax - xmin) * pw, Y = y => m.h + ph - (y - ymin) / (ymax - ymin) * ph;
    ctx.font = "20px " + css("--font-body"); ctx.textBaseline = "middle";
    const sx = pas(xmax - xmin), sy = pas(ymax - ymin);
    ctx.strokeStyle = line; ctx.lineWidth = 1; ctx.fillStyle = mut;
    for (let v = Math.ceil(xmin / sx) * sx; v <= xmax + 1e-9; v += sx) { ctx.beginPath(); ctx.moveTo(X(v), m.h); ctx.lineTo(X(v), m.h + ph); ctx.stroke(); ctx.textAlign = "center"; ctx.fillText(fx(v, sx < 1 ? (sx < .1 ? 2 : 1) : 0), X(v), m.h + ph + 18); }
    for (let v = Math.ceil(ymin / sy) * sy; v <= ymax + 1e-9; v += sy) { ctx.beginPath(); ctx.moveTo(m.g, Y(v)); ctx.lineTo(m.g + pw, Y(v)); ctx.stroke(); ctx.textAlign = "right"; ctx.fillText(fx(v, sy < 1 ? (sy < .1 ? 2 : 1) : 0), m.g - 8, Y(v)); }
    ctx.strokeStyle = ink; ctx.lineWidth = 1.6; ctx.beginPath(); ctx.moveTo(m.g, Y(Math.max(ymin, Math.min(ymax, 0)))); ctx.lineTo(m.g + pw, Y(Math.max(ymin, Math.min(ymax, 0)))); ctx.moveTo(X(Math.max(xmin, 0)), m.h); ctx.lineTo(X(Math.max(xmin, 0)), m.h + ph); ctx.stroke();
    ctx.fillStyle = ink; ctx.font = "700 20px " + css("--font-body"); ctx.textAlign = "right"; ctx.fillText(cfg.xlab || "", m.g + pw, m.h + ph + 40); ctx.textAlign = "left"; ctx.fillText(cfg.ylab || "", 8, 16);
    ctx.save(); ctx.beginPath(); ctx.rect(m.g, m.h - 4, pw, ph + 8); ctx.clip();
    cfg.series.forEach(s => {
      ctx.strokeStyle = ctx.fillStyle = s.c || ink; ctx.lineWidth = s.w || 2.5; ctx.setLineDash(s.dash || []);
      if (s.f) { ctx.beginPath(); for (let i = 0; i <= 300; i++) { const x = xmin + (xmax - xmin) * i / 300, y = s.f(x); i ? ctx.lineTo(X(x), Y(y)) : ctx.moveTo(X(x), Y(y)); } ctx.stroke(); }
      else if (s.points) s.pts.forEach(([x, y]) => { ctx.beginPath(); if (s.croix) { ctx.lineWidth = 2; ctx.moveTo(X(x) - 5, Y(y) - 5); ctx.lineTo(X(x) + 5, Y(y) + 5); ctx.moveTo(X(x) - 5, Y(y) + 5); ctx.lineTo(X(x) + 5, Y(y) - 5); ctx.stroke(); } else { ctx.arc(X(x), Y(y), s.r || 4, 0, 7); ctx.fill(); } });
      else { ctx.beginPath(); s.pts.forEach(([x, y], i) => i ? ctx.lineTo(X(x), Y(y)) : ctx.moveTo(X(x), Y(y))); ctx.stroke(); }
      ctx.setLineDash([]);
    });
    if (cfg.curseur != null) { ctx.strokeStyle = css("--amber"); ctx.lineWidth = 2; ctx.setLineDash([6, 5]); ctx.beginPath(); ctx.moveTo(X(cfg.curseur), m.h); ctx.lineTo(X(cfg.curseur), m.h + ph); ctx.stroke(); ctx.setLineDash([]); }
    ctx.restore();
    /* légende */
    ctx.font = "18px " + css("--font-body"); ctx.textAlign = "left";
    const noms = cfg.series.filter(s => s.nom); let lx = W - m.d - noms.reduce((a, s) => a + ctx.measureText(s.nom).width + 44, 0) + 20;
    ctx.font = "700 20px " + css("--font-body"); lx = Math.max(lx, 16 + ctx.measureText(cfg.ylab || "").width + 24); ctx.font = "18px " + css("--font-body");
    noms.forEach(s => { ctx.fillStyle = s.c || ink; ctx.fillRect(lx, 14, 16, 4); ctx.fillStyle = ink; ctx.fillText(s.nom, lx + 22, 16); lx += ctx.measureText(s.nom).width + 44; });
  }
  return { trace };
}
/* régression polynomiale (moindres carrés), degré 1 ou 2, avec ou sans constante */
function ajuste(pts, deg, constante = true) {
  const base = x => { const b = []; if (constante) b.push(1); b.push(x); if (deg === 2) b.push(x * x); return b; };
  const n = base(0).length, A = Array.from({ length: n }, () => Array(n + 1).fill(0));
  pts.forEach(([x, y]) => { const b = base(x); for (let i = 0; i < n; i++) { for (let j = 0; j < n; j++) A[i][j] += b[i] * b[j]; A[i][n] += b[i] * y; } });
  for (let i = 0; i < n; i++) { let p = i; for (let k = i + 1; k < n; k++) if (Math.abs(A[k][i]) > Math.abs(A[p][i])) p = k; [A[i], A[p]] = [A[p], A[i]];
    for (let k = 0; k < n; k++) if (k !== i) { const f = A[k][i] / A[i][i]; for (let j = i; j <= n; j++) A[k][j] -= f * A[i][j]; } }
  const c = A.map((r, i) => r[n] / r[i]); const res = { c0: constante ? c[0] : 0, c1: c[constante ? 1 : 0], c2: deg === 2 ? c[constante ? 2 : 1] : 0 };
  res.f = x => res.c0 + res.c1 * x + res.c2 * x * x; return res;
}

/* ---------------------------------------------------------------------
   PROTOCOLE (étapes de la fiche, cochées automatiquement)
   --------------------------------------------------------------------- */
function Protocole(etapes) {
  const ol = $("#proto");
  function rendu() {
    const etats = etapes.map(e => { const sous = (e.sous || []).map(x => !!x.ok()); return { ok: e.sous ? sous.every(Boolean) : !!e.ok(), sous }; });
    const courant = etats.findIndex(x => !x.ok);

      const st = etats[k], cls = st.ok ? "fait" : k === courant ? "cur" : "";

      const part = e.partie ? `<li class="partie">${e.partie}</li>` : "";
      return `${part}<li class="${cls}"><span class="q">${e.q}</span><span class="tx">${e.html}${sous}</span><span class="st" aria-label="${st.ok ? "fait" : "à faire"}">${st.ok ? "✓" : k === courant ? "➜" : ""}</span></li>`;
    }).join("");
    const c = $("#protocount"); if (c) c.textContent = `${etats.filter(x => x.ok).length}/${etapes.length} étapes`;
  }

  ["click", "change", "pointerup", "input"].forEach(ev => document.addEventListener(ev, plusTard));
  setInterval(rendu, 1500); rendu();
  LAB.protoRendu = rendu;
}

function lancer() {
  if (!V || enCours) return; if (!LAB.PRED.peutLancer()) return;
  const r = reg(), P = PART[r.part], B = bilan(r); geo = construire(r); let phi = 0; phase(geo, phi);
  const g = geo, vmax = Math.max(B.vf, 1), K = 2.6 / Math.max(vmax, 1e5) * (REDUCED ? 60 : 1);
  let x = 0, k = 1;
  const Wg = n => P.q * (pot(n - 1, r.sync ? n - 1 : 0, r.U) - pot(n, r.sync ? n - 1 : 0, r.U));
  say(`Lancer : ${P.nom}, U<sub>AB</sub> = ${$("#oU").textContent}, d = ${fx(r.d, 1)} m, ${r.N} étage${r.N > 1 ? "s" : ""}${r.N > 1 ? (r.sync ? ", tension inversée à chaque passage" : ", sans inversion") : ""}…`);
  if (B.stop && B.stop.k === 1) { placer(r, 0, 0, Math.sign(Wg(1))); fin(r, B); return; }
  enCours = true;
  stop = V.anime(dt => {
    const gp = g.gaps[k - 1]; let v, Fd = 0, enGap = false, f = 0;
    if (gp && x >= gp.x0) { enGap = true; f = Math.min(1, (x - gp.x0) / (gp.x1 - gp.x0)); const W = Wg(k); v = Math.sqrt(Math.max(0, 2 * (B.ec[k - 1] + W * f) / P.m)); Fd = Math.sign(W); }
    else v = B.v[Math.min(k - 1, B.v.length - 1)];
    if (enGap && B.stop && B.stop.k === k && f >= B.stop.f - 1e-9) { placer(r, gp.x0 + B.stop.f * (gp.x1 - gp.x0), 0, Fd); fin(r, B); return false; }
    placer(r, x, v, Fd);
    $("#hud").innerHTML = `${enGap ? "intervalle " + k + " : champ E⃗, accélération" + (Fd < 0 ? " négative (freinage)" : "") : k > r.N ? "sortie" : "dans un tube : pas de champ, v constante"} · v = ${fr(v, 3)} m·s<sup>−1</sup> · E<sub>c</sub> = ${fr(.5 * P.m * v * v, 3)} J`;
    x += Math.max(K * v, .35) * Math.min(dt, .05);
    const xs = gp && B.stop && B.stop.k === k ? gp.x0 + B.stop.f * (gp.x1 - gp.x0) : null;
    if (xs !== null && x >= xs) x = xs; else if (gp && x >= gp.x1) k++;
    /* la tension s'inverse pendant que la particule traverse un tube */
    if (r.sync && k <= r.N && phi < k - 1 && x >= g.el[k - 1].cx) { phi = k - 1; phase(g, phi); }
    if (x >= g.fin) { fin(r, B); return false; }
    return true;
  });
}

}
