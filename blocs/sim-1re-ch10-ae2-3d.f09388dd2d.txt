function solvePot() {
  const V = new Float32Array((GX + 1) * (GZ + 1)), fix = new Int8Array(V.length), I = (i, j) => j * (GX + 1) + i;
  for (let j = 0; j <= GZ; j++) { const Z = -8 + j * H10; if (Math.abs(Z) <= 5) { fix[I(14, j)] = 1; V[I(14, j)] = UPN; fix[I(34, j)] = 1; V[I(34, j)] = 0; } }
  for (let i = 0; i < V.length; i++) if (!fix[i]) V[i] = UPN / 2;
  for (let it = 0; it < 1800; it++) for (let j = 0; j <= GZ; j++) for (let i = 0; i <= GX; i++) {
    const k = I(i, j); if (fix[k]) continue;
    const l = V[I(i > 0 ? i - 1 : 1, j)], r = V[I(i < GX ? i + 1 : GX - 1, j)], d = V[I(i, j > 0 ? j - 1 : 1)], u = V[I(i, j < GZ ? j + 1 : GZ - 1)];
    V[k] += 1.9 * ((l + r + d + u) / 4 - V[k]);
  }
  return V;
}
function potAt(X, Z) {
  const fx = Math.min(GX - 1e-6, Math.max(0, (X + 12) / H10)), fz = Math.min(GZ - 1e-6, Math.max(0, (Z + 8) / H10)), i = Math.floor(fx), j = Math.floor(fz), a = fx - i, b = fz - j, I = (i, j) => j * (GX + 1) + i;
  return POT[I(i, j)] * (1 - a) * (1 - b) + POT[I(i + 1, j)] * a * (1 - b) + POT[I(i, j + 1)] * (1 - a) * b + POT[I(i + 1, j + 1)] * a * b;
}
