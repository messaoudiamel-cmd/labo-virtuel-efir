function makeChrono(parent, y = .002) {
  const g = new THREE.Group(); parent.add(g);
  const vec = new THREE.Group(); parent.add(vec);
  const C = {
    g, vec, pts: [],
    clear() { g.clear(); vec.clear(); this.pts = []; },
    add(p, i, showLabel = true) {
      const d = new THREE.Mesh(new THREE.SphereGeometry(.0045, 12, 8), new THREE.MeshBasicMaterial({ color: 0x1A237E })); d.position.set(p.x, y + .003, p.z); g.add(d);
      if (showLabel) { const l = label('M' + i, { size: 22, scale: .00045, bg: 'rgba(255,255,255,.8)' }); l.position.set(p.x, y + .02, p.z - .012); g.add(l); }
      this.pts.push(p.clone());
    },
    clearVec() { vec.clear(); },
    arrow(from, v, color, txt, dashed) {
      const L = v.length(); if (L < 1e-5) { const d = new THREE.Mesh(new THREE.RingGeometry(.006, .009, 20), new THREE.MeshBasicMaterial({ color, side: THREE.DoubleSide })); d.rotation.x = -Math.PI / 2; d.position.set(from.x, y + .004, from.z); vec.add(d); if (txt) { const l = label(txt + ' = 0', { size: 22, scale: .0005, color: '#' + color.toString(16).padStart(6, '0') }); l.position.set(from.x + .03, y + .03, from.z); vec.add(l); } return; }
      const dir = v.clone().normalize(), o = new THREE.Vector3(from.x, y + .004, from.z);
      if (dashed) { const n = Math.max(2, Math.round(L / .012)); for (let k = 0; k < n; k += 2) { const a = o.clone().addScaledVector(dir, L * k / n), b = o.clone().addScaledVector(dir, Math.min(L * (k + 1) / n, L - .008)); vec.add(new THREE.Line(new THREE.BufferGeometry().setFromPoints([a, b]), new THREE.LineBasicMaterial({ color }))); } const head = new THREE.ArrowHelper(dir, o.clone().addScaledVector(dir, L - .009), .009, color, .009, .006); vec.add(head); }
      else vec.add(new THREE.ArrowHelper(dir, o, L, color, Math.min(.014, L * .35), Math.min(.009, L * .25)));
      if (txt) { const l = label(txt, { size: 24, scale: .00038, color: '#' + color.toString(16).padStart(6, '0'), bg: 'rgba(255,255,255,.6)' }); const m = o.clone().addScaledVector(dir, L * .55), sd = dashed ? -1 : 1; l.position.set(m.x - sd * dir.z * .016, y + .012, m.z + sd * dir.x * .016); vec.add(l); }
    },
  };
  return C;
}
