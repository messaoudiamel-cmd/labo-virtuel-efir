const GG14 = 9.81;
// données brutes [{t, x, y}] → ajoute v (dérivée centrée, comme un logiciel de pointage) et les énergies
function energize(raw, m) {
  return raw.map((p, i) => {
    const a = raw[Math.max(0, i - 1)], b = raw[Math.min(raw.length - 1, i + 1)], dt = b.t - a.t;
    const v = Math.hypot(b.x - a.x, b.y - a.y) / dt;
    const Ec = .5 * m * v * v, Ep = m * GG14 * p.y; return { ...p, v, Ec, Ep, Em: Ec + Ep };
  }).slice(1, -1);
}
function energyChart(cv, data, o = {}) {
  if (!data.length) { drawChart(cv, { xmax: o.tmax || 2, ymax: o.ymax || .1, xlabel: 't (s)', ylabel: 'Énergie (J)', pts: [] }); return; }
  const tmax = o.tmax || Math.ceil(data[data.length - 1].t * 5) / 5, ymax = o.ymax || Math.max(...data.map(d => d.Em)) * 1.25;
  drawChart(cv, { xmax: tmax, ymax, xlabel: 't (s)', ylabel: 'Énergie (J)', pts: [], curves: [{ pts: data.map(d => [d.t, d.Ec]), color: '#E65100' }, { pts: data.map(d => [d.t, d.Ep]), color: '#2F5DA8' }, { pts: data.map(d => [d.t, d.Em]), color: '#2e7d32' }], xfmt: x => x.toFixed(2).replace('.', ','), yfmt: y => y.toPrecision(2).replace('.', ',') });
}
const LEG14 = '<span style="color:#E65100">■ ℰc</span> &nbsp; <span style="color:#2F5DA8">■ ℰp</span> &nbsp; <span style="color:#2e7d32">■ ℰm = ℰc + ℰp</span>';
function energyTable(data, n = 3) {
  const f = (v, k) => v.toFixed(k).replace('.', ',');
  return '<thead><tr><th>t (s)</th><th>x (m)</th><th>y (m)</th><th>v (m·s⁻¹)</th><th>ℰc (J)</th><th>ℰp (J)</th><th>ℰm (J)</th></tr></thead><tbody>' + data.map(d => `<tr><td>${f(d.t, 2)}</td><td>${f(d.x, 3)}</td><td>${f(d.y, 3)}</td><td>${f(d.v, 3)}</td><td>${f(d.Ec, n)}</td><td>${f(d.Ep, n)}</td><td>${f(d.Em, n)}</td></tr>`).join('') + '</tbody>';
}

function simulate20(mil) {
  const rho = MIL[mil].rho; let y = Y0B, v = 0, t = 0, next = 0; const h = .0005, out = [];
  const acc = v => (MB * GG14 - rho * VB * GG14 - .5 * rho * .5 * AB * v * Math.abs(v)) / MB; // v positive vers le bas
  while (y > RB + .01 && t < 3) { if (t >= next - 1e-9) { const nz = Math.sin(t * 77.7) * .0008; out.push({ t: +next.toFixed(2), x: 0, y: +(y + nz).toFixed(4) }); next += DTB; } const a = acc(v); v += a * h; y -= v * h; t += h; }
  return out;
}
