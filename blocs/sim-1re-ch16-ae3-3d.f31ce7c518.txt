const AXY = .15, BX = p => (p - 60) / 100, LD16 = 4; // hauteur de l'axe (m) ; conversion cm → m ; diamètre utile de la lentille (cm)
function conj(obj, lens, f) { // renvoie {xA, xAp, g, real}
  const xA = obj - lens, inv = 1 / f + 1 / xA;
  if (Math.abs(inv) < 1e-9) return { xA, xAp: Infinity, g: Infinity, real: false };
  const xAp = 1 / inv; return { xA, xAp, g: xAp / xA, real: xAp > 0 };
}
function drawF(c, cx, cy, hpx, flip, col) {
  c.save(); c.translate(cx, cy); if (flip) c.rotate(Math.PI); c.fillStyle = col;

}
