// Ch. 4 · AE1 · Titrage colorimétrique (ions fer (II) d'un produit anti-chlorose par les ions permanganate)
// ====================================================================
// Module commun : titrage direct avec repérage colorimétrique de l'équivalence
// ====================================================================
function makeTitrTP(cfg) {
  let bu, er, stir, pip, S = null, flash = 0;
  const fresh = () => ({ pip: false, ind: !cfg.indicator, fill: false, zero: false, V: 0, stir: false, crossed: null, n: 1, results: [] });
  const fmtV = v => v.toFixed(2).replace('.', ',');
  function colorAt(V) {
    const VE = cfg.VE; const after = V >= VE - 1e-9;
    if (after) { const k = Math.min(1, (V - VE + .05) / .4); return { c: new THREE.Color(cfg.colBefore).lerp(new THREE.Color(cfg.colAfter), k), o: cfg.opBefore + (cfg.opAfter - cfg.opBefore) * k }; }
    return { c: new THREE.Color(cfg.colBefore), o: cfg.opBefore };
  }
  function paint() {
    if (!er) return; const vol = cfg.V0 + (cfg.water || 0) + S.V; er.setVolume(S.pip ? vol : 0);
    const { c, o } = colorAt(S.V); er.setColor(c, o);
    if (flash > 0 && S.V < cfg.VE) { er.mat.color.lerp(new THREE.Color(cfg.colFlash || cfg.colAfter), flash); er.mat.opacity = Math.max(er.mat.opacity, o + .25 * flash); }
    if (cfg.precip) { const k = Math.min(S.V, cfg.VE) / cfg.VE; er.mat.color.lerp(new THREE.Color(0xffffff), .18 * k); er.mat.opacity = Math.max(er.mat.opacity, .45 + .4 * k); }
    if (S.fill) bu.set(S.zero ? S.V : 0, S.zero ? 0 : 1.2); else bu.empty();
    const r = $('#tV'); if (r) r.innerHTML = S.zero ? `V versé = <b>${fmtV(S.V)} mL</b>` : '—';
  }
  const TP = Object.assign({
    views: { ensemble: [[0, .85, 1.55], [0, .3, 0]] },
    viewbar: [['Zoom burette (lecture)', () => { const y = bu ? .32 + .5 * .95 - .45 * S.V / 25 : .6; flyTo([.05, y + .02, .35], [0, y, 0], .9); }], ['Zoom erlenmeyer', () => flyTo([.1, .35, .55], [0, .12, 0])]],
    build() {
      stir = makeStirrer(); scene.add(stir.group);
      er = makeErlen(125, 'Erlenmeyer'); er.group.position.y = stir.top; scene.add(er.group);
      bu = makeBurette(25); scene.add(bu.group); bu.setColor(cfg.titrantColor, cfg.titrantOp);
      pip = makeVessel({ cap: 20, r: .008, hgt: .3, kind: 'becher', label: 'Pipette jaugée 20,0 mL' }); pip.group.position.set(.42, .02, .1); scene.add(pip.group);
      const fb = makeBottle(cfg.titrantBottle, cfg.titrantColor, { cap: 0x6a1b9a, brown: !!cfg.brown }); fb.position.set(-.5, 0, -.25); scene.add(fb);
      const ft = makeBottle(cfg.titratedBottle, cfg.titratedColor ?? cfg.colBefore, { cap: 0x2e7d32 }); ft.position.set(.6, 0, -.25); scene.add(ft);
      if (cfg.indicator) { const fi = makeBottle(cfg.indicator, cfg.indicatorColor || 0xffeb3b, { cap: 0xf9a825 }); fi.scale.setScalar(.6); fi.position.set(.85, 0, .15); scene.add(fi); }
      const bid = new THREE.Mesh(new THREE.BoxGeometry(.26, .32, .18), new THREE.MeshStandardMaterial({ color: 0xffb74d })); bid.position.set(-.8, .16, .15); scene.add(bid);
      const bl = label('Bidon de\nrécupération', { size: 30, scale: .0011 }); bl.position.set(-.8, .43, .15); scene.add(bl);
      LAB.tick.push(dt => { if (flash > 0) { flash = Math.max(0, flash - dt / (S.V > cfg.VE * .9 ? 2.2 : .8)); paint(); } });
      this.resetScene(); this.buildPanel();
    },
    resetScene(keep) { const res = keep && S ? S.results : []; const n = keep && S ? S.n + 1 : 1; S = fresh(); S.results = res; S.n = n; flash = 0; if (stir) stir.on = false; if (er) paint(); if ($('#tV')) this.refreshPanel(); },
    buildPanel() {
      $('#actions').append(
        h('h3', {}, '1. Préparer'),
        h('div', { class: 'row' },
          h('button', { id: 'tPip', onclick: () => act({ t: 'pip' }) }, `Prélever ${cfg.V0.toFixed(1).replace('.', ',')} mL de ${cfg.titratedShort} (pipette jaugée)`),
          cfg.indicator ? h('button', { id: 'tInd', onclick: () => act({ t: 'ind' }) }, 'Ajouter quelques gouttes de ' + cfg.indicatorShort) : null,
          h('button', { id: 'tStir', onclick: () => act({ t: 'stir' }) }, 'Mettre l\'agitation'),
          h('button', { id: 'tFill', onclick: () => act({ t: 'fill' }) }, 'Remplir la burette (' + cfg.titrantShort + ')'),
          h('button', { id: 'tZero', onclick: () => act({ t: 'zero' }) }, 'Ajuster au zéro')),
        h('h3', {}, '2. Titrer'),
        h('div', { class: 'row' }, ...[[5, '+5 mL'], [1, '+1 mL'], [.5, '+0,5 mL'], [.1, '+0,1 mL'], [.05, '1 goutte (0,05 mL)']].map(([d, t]) => h('button', { 'data-add': d, onclick: () => act({ t: 'add', v: d }) }, t))),
        h('div', { id: 'tV', style: 'margin:6px 0' }, '—'),
        h('h3', {}, '3. Recommencer / ranger'),
        h('div', { class: 'row' }, h('button', { onclick: () => act({ t: 'vider', where: 'bidon' }) }, '🛢 Vider l\'erlenmeyer dans le bidon (nouveau titrage)'), h('button', { class: 'warn', onclick: () => act({ t: 'vider', where: 'evier' }) }, '🚰 Vider à l\'évier')),
        h('div', { class: 'note', id: 'tRes', style: 'margin-top:6px' }));
      this.refreshPanel();
    },
    refreshPanel() {
      const en = (s, ok) => $$(s).forEach(b => b.disabled = BUSY || !ok);
      en('#tPip', !S.pip); en('#tInd', S.pip && !S.ind); en('#tStir', S.pip && !S.stir); en('#tFill', !S.fill); en('#tZero', S.fill && !S.zero);
      en('[data-add]', S.pip && S.ind && S.zero && S.V < 25);
      $('#tRes').innerHTML = S.results.length ? '<b>Mes titrages :</b><br>' + S.results.map((r, i) => `Titrage ${i + 1} : changement de couleur pour V = ${fmtV(r)} mL`).join('<br>') : '';
      paint();
    },
    refresh() { this.refreshPanel(); },
    async doAction(a) {
      if (a.t === 'pip') {
        pip.setColor(cfg.titratedColor ?? cfg.colBefore, .35); await tween(.7, p => pip.setVolume(20 * p));
        const p0 = pip.group.position.clone(), p1 = new THREE.Vector3(0, stir.top + er.top() + .02, 0);
        await tween(.7, p => pip.group.position.lerpVectors(p0, p1, ease(p)));
        S.pip = true; await tween(.9, p => { pip.setVolume(20 * (1 - p)); er.setVolume((cfg.V0 + (cfg.water || 0)) * p); });
        await tween(.6, p => pip.group.position.lerpVectors(p1, p0, ease(p)));
        say(`${cfg.V0.toFixed(1).replace('.', ',')} mL de ${cfg.titratedShort} dans l'erlenmeyer.`); paint(); step('prep'); return;
      }
      if (a.t === 'ind') { S.ind = true; await dropInto({ group: er.group, top: () => er.top(), R: 0, level: .03 }, cfg.indicatorColor || 0xffeb3b, 3); say('Quelques gouttes de ' + cfg.indicatorShort + ' ajoutées : le mélange devient ' + cfg.indicatorEffect + '.'); paint(); return; }
      if (a.t === 'stir') { S.stir = true; stir.on = true; say('Agitation magnétique en marche.'); return; }
      if (a.t === 'fill') { S.fill = true; await tween(.8, p => bu.set(0, 1.2 * p)); say('Burette remplie au-dessus du zéro avec ' + cfg.titrantShort + '.'); return; }
      if (a.t === 'zero') { await tween(.6, p => bu.set(0, 1.2 * (1 - p))); S.zero = true; say('Niveau ajusté au zéro : la burette affiche 0,00 mL.'); step('burette'); paint(); return; }
      if (a.t === 'add') {
        const v0 = S.V, v1 = Math.min(25, +(S.V + a.v).toFixed(2));
        if (!S.stir) say('<b>Conseil :</b> mets l\'agitation pour bien mélanger.');
        await tween(Math.min(1.2, .25 + a.v * .3), p => { S.V = v0 + (v1 - v0) * p; paint(); });
        S.V = v1; if (S.V < cfg.VE) flash = Math.min(1, .5 + S.V / cfg.VE * .5);
        paint();
        if (S.V >= cfg.VE && S.crossed == null) {
          S.crossed = S.V; S.results.push(S.V);
          const msg = `Titrage ${S.results.length} : ${cfg.changeText} pour V = ${fmtV(S.V)} mL.`; observe(msg); say(msg);
          step('titrage'); if (Math.abs(S.V - cfg.VE) <= .1) step('precis');
        } else if (S.V < cfg.VE) say(`V = ${fmtV(S.V)} mL : ${S.V > cfg.VE * .85 ? cfg.nearText : cfg.beforeText}`);
        else say(`V = ${fmtV(S.V)} mL : ${cfg.afterText}`);
        return;
      }
      if (a.t === 'vider') {
        if (!S.pip) { say('L\'erlenmeyer est vide.'); return false; }
        if (a.where === 'evier') { ST.evier++; observe('Vidage de l\'erlenmeyer à l\'évier (interdit).'); say('<b>Interdit !</b> ' + cfg.wasteText); }
        else { ST.bidon++; step('vider'); say('Erlenmeyer vidé dans le bidon et rincé. Tu peux faire un nouveau titrage (au goutte à goutte près de l\'équivalence).'); }
        await tween(.6, p => er.setVolume(er.vol * (1 - p)));
        this.resetScene(true); return;
      }
    },
    async demo() {
      say('Démonstration : préparation du titrage.');
      await act({ t: 'pip' }, { record: false }); if (cfg.indicator) await act({ t: 'ind' }, { record: false });
      await act({ t: 'stir' }, { record: false }); await act({ t: 'fill' }, { record: false }); await act({ t: 'zero' }, { record: false });
      flyTo([.1, .35, .55], [0, .12, 0]);
      for (let i = 0; i < 3; i++) { await act({ t: 'add', v: 1 }, { record: false }); await wait(500); }
      say('Près de l\'équivalence, on verse goutte à goutte. À toi !'); await wait(1500);
    },
  }, cfg.tp);
  return TP;
}
