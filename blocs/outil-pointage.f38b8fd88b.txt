    <details class="aide" style="margin-top:16px"><summary>❓ Aide : exploiter un pointage</summary><ul>
      <li>Le bouton « Vitesses » ajoute <code>vx = deriv(x, t)</code>, <code>vy = deriv(y, t)</code> et <code>v = sqrt(vx^2 + vy^2)</code>. La vitesse au point n est calculée avec les points n−1 et n+1 : elle n'existe donc ni au premier ni au dernier point.</li>
      <li>Le bouton « Énergies » ajoute la masse <code>m</code> et <code>g</code> en paramètres : modifiez la masse avant d'interpréter.</li>
      <li>Pour modéliser, choisissez la grandeur et un modèle : les paramètres sont ajustés par la méthode des moindres carrés et R² indique la qualité de l'accord (proche de 1 : bon accord).</li>
      <li>Formule libre : écrivez le modèle avec la grandeur en abscisse et des lettres pour les paramètres, par exemple <code>a*t^2+b*t+c</code> ou <code>A*exp(-t/tau)</code>.</li>
      <li>Pour coller des données d'un tableur, copiez les colonnes (avec la ligne des noms) et utilisez « Coller des données ».</li>
    </ul></details>

function fmt(v, p = 4) {
  if (v === null || v === undefined || !isFinite(v)) return "";
  if (v === 0) return "0";
  const a = Math.abs(v); let s;
  if (a >= 1e5 || a < 1e-3) s = v.toExponential(p - 1).replace(/\.?0+e/, "e").replace("e+", "e");
  else s = String(Number(v.toPrecision(p)));
  return s.replace(".", ",");
}

const FN = { sqrt: Math.sqrt, exp: Math.exp, ln: Math.log, log: Math.log10, sin: Math.sin, cos: Math.cos, tan: Math.tan, abs: Math.abs, asin: Math.asin, acos: Math.acos, atan: Math.atan };
function lexer(src) {
  const T = []; let i = 0;
  while (i < src.length) {
    const c = src[i];
    if (/\s/.test(c)) { i++; continue; }
    let m = /^(\d+\.?\d*|\.\d+)([eE][+-]?\d+)?/.exec(src.slice(i));
    if (m) { T.push({ k: "n", v: Number(m[0]) }); i += m[0].length; continue; }
    m = /^[A-Za-z_][A-Za-z0-9_]*/.exec(src.slice(i));
    if (m) { T.push({ k: "id", v: m[0] }); i += m[0].length; continue; }
    if ("+-*/^(),".includes(c)) { T.push({ k: c }); i++; continue; }
    if (c === "²") { T.push({ k: "^" }, { k: "n", v: 2 }); i++; continue; }
    throw new Error("Caractère non reconnu : « " + c + " »");
  }
  return T;
}
function parse(src) {
  const T = lexer(src); let p = 0;
  const voit = k => T[p] && T[p].k === k, prend = k => { if (!voit(k)) throw new Error(k === ")" ? "Parenthèse fermante manquante" : "Formule incomplète"); return T[p++]; };
  function expr() { let a = terme(); while (voit("+") || voit("-")) { const o = T[p++].k; a = { o, a, b: terme() }; } return a; }
  function terme() { let a = unaire(); while (voit("*") || voit("/")) { const o = T[p++].k; a = { o, a, b: unaire() }; } return a; }
  function unaire() { if (voit("-")) { p++; return { o: "neg", a: unaire() }; } if (voit("+")) { p++; return unaire(); } return puiss(); }
  function puiss() { const a = prim(); if (voit("^")) { p++; return { o: "^", a, b: unaire() }; } return a; }
  function prim() {
    if (voit("n")) return { o: "n", v: T[p++].v };
    if (voit("(")) { p++; const e = expr(); prend(")"); return e; }
    if (voit("id")) { const n = T[p++].v;
      if (voit("(")) { p++; const args = [expr()]; while (voit(",")) { p++; args.push(expr()); } prend(")");
        if (n === "deriv") { if (args.length !== 2) throw new Error("deriv s'écrit deriv(grandeur, variable)"); return { o: "deriv", a: args[0], b: args[1] }; }
        if (!FN[n]) throw new Error("Fonction inconnue : " + n); if (args.length !== 1) throw new Error(n + " prend un seul argument"); return { o: "f", f: n, a: args[0] }; }
      return { o: "id", n }; }
    throw new Error(p >= T.length ? "Formule incomplète" : "Formule mal écrite");
  }
  if (!T.length) throw new Error("Formule vide");
  const e = expr(); if (p < T.length) throw new Error("Formule mal écrite près de « " + (T[p].v ?? T[p].k) + " »"); return e;
}

/* évaluation vectorielle : une valeur est un nombre ou un tableau (une colonne) */
function map2(a, b, f) {
  const A = Array.isArray(a), B = Array.isArray(b);
  if (!A && !B) return f(a, b);
  const n = A ? a.length : b.length, r = new Array(n);
  for (let i = 0; i < n; i++) r[i] = f(A ? a[i] : a, B ? b[i] : b);
  return r;
}
const map1 = (a, f) => Array.isArray(a) ? a.map(f) : f(a);
function evalue(e, env) {
  switch (e.o) {
    case "n": return e.v;
    case "id": if (e.n === "pi") return Math.PI; if (!(e.n in env)) throw new Error("Grandeur inconnue : " + e.n); return env[e.n];
    case "neg": return map1(evalue(e.a, env), x => -x);
    case "+": return map2(evalue(e.a, env), evalue(e.b, env), (x, y) => x + y);
    case "-": return map2(evalue(e.a, env), evalue(e.b, env), (x, y) => x - y);
    case "*": return map2(evalue(e.a, env), evalue(e.b, env), (x, y) => x * y);
    case "/": return map2(evalue(e.a, env), evalue(e.b, env), (x, y) => x / y);
    case "^": return map2(evalue(e.a, env), evalue(e.b, env), (x, y) => Math.pow(x, y));
    case "f": return map1(evalue(e.a, env), FN[e.f]);
    case "deriv": { const A = evalue(e.a, env), V = evalue(e.b, env);
      if (!Array.isArray(A) || !Array.isArray(V)) throw new Error("deriv s'applique à des colonnes du tableau");
      const n = A.length, d = new Array(n).fill(NaN);
      for (let i = 1; i < n - 1; i++) d[i] = (A[i + 1] - A[i - 1]) / (V[i + 1] - V[i - 1]);
      return d; }
  }
  throw new Error("Formule mal écrite");
}

function resout(A, b) { const n = b.length, M = A.map((r, i) => r.concat([b[i]]));
  for (let c = 0; c < n; c++) { let p = c; for (let r = c + 1; r < n; r++) if (Math.abs(M[r][c]) > Math.abs(M[p][c])) p = r; [M[c], M[p]] = [M[p], M[c]]; if (Math.abs(M[c][c]) < 1e-300) return null;
    for (let r = 0; r < n; r++) if (r !== c) { const f = M[r][c] / M[c][c]; for (let k = c; k <= n; k++) M[r][k] -= f * M[c][k]; } }
  return M.map((r, i) => r[n] / r[i]); }

    if (!ok) break;
  }
  return { p, s: cur.s };
}
function initiales(type, noms, X, Y) {
  const n = X.length, xm = Math.min(...X), xM = Math.max(...X), ym = Math.min(...Y), yM = Math.max(...Y), moy = Y.reduce((a, b) => a + b, 0) / n, d = { };
  noms.forEach(k => d[k] = 1);
  if (type === "exd") { d.B = Y[n - 1]; d.A = Y[0] - Y[n - 1] || 1; d.tau = (xM - xm) / 3 || 1; }
  if (type === "exc") { d.A = Y[n - 1] || 1; d.tau = (xM - xm) / 3 || 1; }
  if (type === "pui") { const P = X.map((x, i) => [x, Y[i]]).filter(([x, y]) => x > 0 && y > 0).map(([x, y]) => [Math.log(x), Math.log(y)]);
    if (P.length >= 2) { const mx = P.reduce((s, p) => s + p[0], 0) / P.length, my = P.reduce((s, p) => s + p[1], 0) / P.length; let a = 0, b = 0; P.forEach(([u, v]) => { a += (u - mx) * (v - my); b += (u - mx) ** 2; }); d.n = b ? a / b : 1; d.a = Math.exp(my - d.n * mx); } }
  if (type === "sin") { d.B = moy; d.A = (yM - ym) / 2 || 1; const z = []; for (let i = 1; i < n; i++) if ((Y[i - 1] - moy) * (Y[i] - moy) < 0) z.push(X[i]);
    d.T = z.length >= 2 ? 2 * (z[z.length - 1] - z[0]) / (z.length - 1) : (xM - xm) || 1; d.phi = 0; }
  if (type === "libre") { noms.forEach(k => { if (/^(tau|T)$/.test(k)) d[k] = (xM - xm) / 3 || 1; }); }
  return noms.map(k => d[k]);
}
