const pulse = (x, t, w = .03) => Math.exp(-(((x - C15 * t) / w) ** 2));
function buildMedia() {
  // corde
  { const g = new THREE.Group(); g.position.set(...MED.corde.pos); const n = 120, pts = []; for (let i = 0; i <= n; i++) pts.push(new THREE.Vector3(-LEN15 / 2 + LEN15 * i / n, .1, 0));
    const geo = new THREE.BufferGeometry().setFromPoints(pts); const line = new THREE.Line(geo, new THREE.LineBasicMaterial({ color: 0xc62828 })); g.add(line);
    const tube = new THREE.Mesh(new THREE.TubeGeometry(new THREE.CatmullRomCurve3(pts), 120, .003, 6), new THREE.MeshStandardMaterial({ color: 0xc62828 })); g.add(tube);
    const mk = new THREE.Mesh(new THREE.SphereGeometry(.007, 12, 8), new THREE.MeshStandardMaterial({ color: 0xffeb3b, emissive: 0x332b00 })); g.add(mk);
    for (const x of [-LEN15 / 2 - .01, LEN15 / 2 + .01]) { const p = new THREE.Mesh(new THREE.CylinderGeometry(.006, .006, .14, 8), new THREE.MeshStandardMaterial({ color: 0x546e7a })); p.position.set(x, .07, 0); g.add(p); }

  // ressort
  { const g = new THREE.Group(); g.position.set(...MED.ressort.pos); const n = 40, rings = [];
    for (let i = 0; i <= n; i++) { const r = new THREE.Mesh(new THREE.TorusGeometry(.018, .0018, 6, 24), new THREE.MeshStandardMaterial({ color: i === 24 ? 0xffb300 : 0x78909c, metalness: .7, roughness: .3 })); r.rotation.y = Math.PI / 2; r.position.set(-LEN15 / 2 + LEN15 * i / n, .1, 0); r.x0 = r.position.x; g.add(r); rings.push(r); }
    const l = label('Ressort', { size: 24, scale: .0008 }); l.position.set(0, .2, 0); g.add(l); scene.add(g);
    M15.ressort = { g, lab: l, update(t) { for (const r of rings) { const xi = r.x0 + LEN15 / 2; r.position.x = r.x0 + (t < 0 ? 0 : .012 * Math.exp(-(((xi - C15 * t) / .04) ** 2)) * ((C15 * t - xi) > 0 ? -1 : 1) * Math.min(1, Math.abs(xi - C15 * t) / .02)); } } }; }
  // eau
  { const g = new THREE.Group(); g.position.set(...MED.eau.pos); const geo = new THREE.PlaneGeometry(.5, .35, 90, 64); geo.rotateX(-Math.PI / 2); const base = geo.attributes.position.array.slice();
    const water = new THREE.Mesh(geo, new THREE.MeshStandardMaterial({ color: 0x4fc3f7, roughness: .15, metalness: .1, transparent: true, opacity: .85 })); water.position.y = .03; g.add(water);
    const tray = new THREE.Mesh(new THREE.BoxGeometry(.52, .03, .37), new THREE.MeshStandardMaterial({ color: 0x37474f })); tray.position.y = .0; g.add(tray);
    const cork = new THREE.Mesh(new THREE.CylinderGeometry(.008, .008, .008, 12), new THREE.MeshStandardMaterial({ color: 0xffb300 })); g.add(cork);
    const l = label('Cuve à ondes (surface de l\'eau)', { size: 24, scale: .0008 }); l.position.set(0, .14, -.15); g.add(l); scene.add(g);
    const hAt = (x, z, t) => { if (t < 0) return 0; const r = Math.hypot(x + .15, z); return .018 * Math.exp(-(((r - C15 * .8 * t) / .025) ** 2)) / Math.sqrt(1 + r * 6); };
    M15.eau = { g, lab: l, update(t) { const a = geo.attributes.position.array; for (let i = 0; i < a.length; i += 3) a[i + 1] = base[i + 1] + hAt(base[i], base[i + 2], t); geo.attributes.position.needsUpdate = true; geo.computeVertexNormals(); cork.position.set(.1, .034 + hAt(.1, 0, t), 0); } }; }
  // air
  { const g = new THREE.Group(); g.position.set(...MED.air.pos); const gm = GLASS(); gm.opacity = .15; const tube = new THREE.Mesh(new THREE.CylinderGeometry(.05, .05, LEN15, 32, 1, true), gm); tube.rotation.z = Math.PI / 2; tube.position.y = .08; g.add(tube);
    const N = 420, arr = new Float32Array(N * 3), x0 = []; for (let i = 0; i < N; i++) { const x = -LEN15 / 2 + LEN15 * (i + Math.random()) / N, r = Math.sqrt(Math.random()) * .045, a = Math.random() * 6.283; arr.set([x, .08 + r * Math.cos(a), r * Math.sin(a)], i * 3); x0.push(x); }
    const geo = new THREE.BufferGeometry(); geo.setAttribute('position', new THREE.BufferAttribute(arr, 3)); const pts = new THREE.Points(geo, new THREE.PointsMaterial({ color: 0x1565c0, size: .006 })); g.add(pts);
    const mk = new THREE.Mesh(new THREE.SphereGeometry(.006, 12, 8), new THREE.MeshStandardMaterial({ color: 0xffb300, emissive: 0x332b00 })); g.add(mk);
    const hp = new THREE.Mesh(new THREE.CylinderGeometry(.06, .07, .03, 24), new THREE.MeshStandardMaterial({ color: 0x212121 })); hp.rotation.z = Math.PI / 2; hp.position.set(-LEN15 / 2 - .02, .08, 0); g.add(hp);
    const l = label('Air dans un tube (modèle : molécules)', { size: 24, scale: .0008 }); l.position.set(0, .2, 0); g.add(l); scene.add(g);
    const u = (x, t) => t < 0 ? 0 : .025 * Math.exp(-(((x + LEN15 / 2 - C15 * t) / .04) ** 2));
    M15.air = { g, lab: l, update(t) { const a = geo.attributes.position.array; for (let i = 0; i < N; i++) a[i * 3] = x0[i] + u(x0[i], t); geo.attributes.position.needsUpdate = true; mk.position.set(.05 + u(.05, t), .08, .05); } }; }
}
