/* ===== Cadrage automatique de la scène 3D (commun aux paillasses réalistes) =====
   - facteur(nom) : distance multipliée pour que, dans la vue « ensemble », tout le matériel ET ses étiquettes
     tiennent dans le cadre, quelle que soit la forme du panneau (portable, grand écran, téléphone) ;
     pour les vues rapprochées, simple correction selon le rapport largeur/hauteur.
   - etiquettes(cam, liste) : chaque étiquette reste entièrement visible (elle se décale vers l'intérieur). */
const CADRAGE = (function () {
  const T = window.THREE, v = () => new T.Vector3();
  function boite(scene, exclure) {
    const B = new T.Box3(), b = new T.Box3(), s = v(); scene.updateMatrixWorld(true);
    scene.traverse(o => {
      if (!o.isMesh || o.isSprite) return; let p = o; while (p) { if (p.visible === false || (exclure && exclure(p))) return; p = p.parent; }
      b.setFromObject(o); if (b.isEmpty()) return; b.getSize(s); if (Math.max(s.x, s.z) > 90 || s.y > 140) return; B.union(b);
    });
    return B;
  }
  function pose(c, V, d) { c.position.set(V.cx + d * Math.sin(V.ph) * Math.sin(V.th), V.cy + d * Math.cos(V.ph), V.cz + d * Math.sin(V.ph) * Math.cos(V.th)); c.lookAt(V.cx, V.cy, V.cz); c.updateMatrixWorld(true); }
  function demi(cam, s) { const t = Math.tan(cam.fov * Math.PI / 360); return [.5 * s.scale.x / (t * cam.aspect), .5 * s.scale.y / t]; }
  /* plus petite distance telle que la boîte et les étiquettes tiennent dans le cadre */
  function distance(cam, V, B, etiq) {
    const pts = []; for (let i = 0; i < 8; i++) pts.push([new T.Vector3(i & 1 ? B.max.x : B.min.x, i & 2 ? B.max.y : B.min.y, i & 4 ? B.max.z : B.min.z), .92, .86]);
    (etiq || []).forEach(s => { if (!s.visible) return; const [hx, hy] = demi(cam, s); pts.push([s.getWorldPosition(v()), .97 - .6 * hx, .95 - .6 * hy]); });
    const c = cam.clone(); let lo = 4, hi = 700;
    for (let k = 0; k < 28; k++) {
      const d = (lo + hi) / 2; pose(c, V, d); let ok = true;
      for (const [p, mx, my] of pts) { const q = p.clone().project(c); if (q.z > 1 || Math.abs(q.x) > Math.max(.2, mx) || Math.abs(q.y) > Math.max(.2, my)) { ok = false; break; } }
      if (ok) hi = d; else lo = d;
    }
    return hi;
  }
  function etiquettes(cam, liste, H) {
    const q = v(), f = 1;
    liste.forEach(s => {
      if (!s.visible) return; s.center.set(.5, .5);
      if (!s.userData.base) s.userData.base = [s.scale.x, s.scale.y]; s.scale.set(s.userData.base[0] * f, s.userData.base[1] * f, 1); s.getWorldPosition(q).project(cam); const [hx, hy] = demi(cam, s);
      const mx = .985 - hx, my = .97 - hy;
      if (q.x > mx) s.center.x = Math.min(1.5, .5 + (q.x - mx) / (2 * hx)); else if (q.x < -mx) s.center.x = Math.max(-.5, .5 + (q.x + mx) / (2 * hx));
      if (q.y > my) s.center.y = Math.min(1.5, .5 + (q.y - my) / (2 * hy)); else if (q.y < -my) s.center.y = Math.max(-.5, .5 + (q.y + my) / (2 * hy));
    });
  }
  /* vue d'ensemble : centre la caméra sur le matériel et règle la distance ; renvoie le facteur de distance */
  function ensemble(cam, scene, V, cible, exclure) {
    const B = boite(scene, exclure); if (B.isEmpty()) return 1; const c = B.getCenter(v());
    const W = Object.assign({}, V, { cx: c.x, cy: c.y - (B.max.y - B.min.y) * .03, cz: c.z });
    Object.assign(cible, { cx: W.cx, cy: W.cy, cz: W.cz }); return distance(cam, W, B, null) / V.d;
  }
  return { boite, distance, etiquettes, ensemble };
})();

const SPECTRO = (function () {
  const ESPECES = {
    e133: {
      nom: "Bleu brillant (E133)", court: "E133", unite: "µmol·L⁻¹", f: 1e-6, couleurPercue: "bleu-vert (turquoise)",
      bandes: [[630, 1.30e5, 32], [410, 1.2e4, 30], [570, 2.2e4, 25]], gamme: [1, 2, 3, 4, 5], inconnue: 3.4,
      contexte: "Le sirop de menthe contient le colorant bleu brillant E133. On veut sa concentration dans le sirop dilué 20 fois."
    },
    kmno4: {
      nom: "Permanganate de potassium", court: "MnO₄⁻", unite: "mmol·L⁻¹", f: 1e-3, couleurPercue: "violet-pourpre",
      bandes: [[530, 1.75e3, 26], [525, 7.0e2, 7], [546, 6.2e2, 7], [507, 3.5e2, 7], [566, 2.2e2, 7], [320, 3.0e3, 25]], gamme: [.02, .04, .06, .08, .10], inconnue: .063,
      contexte: "La solution de Dakin (antiseptique) doit sa couleur rose aux ions permanganate MnO₄⁻ ; l'étiquette indique 0,0010 % de KMnO₄ (M = 158,0 g·mol⁻¹). On vérifie cette indication."
    },
    e102: {
      nom: "Tartrazine (E102)", court: "E102", unite: "µmol·L⁻¹", f: 1e-6, couleurPercue: "jaune",
      bandes: [[427, 2.60e4, 38], [260, 1.0e4, 20]], gamme: [5, 10, 15, 20, 25], inconnue: 13.5,
      contexte: "Un sirop de citron contient le colorant jaune tartrazine E102. On veut sa concentration."
    }
  };
  function eps(sp, l) { return ESPECES[sp].bandes.reduce((s, [m, e, w]) => s + e * Math.exp(-.5 * Math.pow((l - m) / w, 2)), 0); }
  function A(sp, C, l, lc) { return eps(sp, l) * (lc || 1) * C * ESPECES[sp].f; }
  function lmax(sp) { let b = 400, m = 0; for (let l = 400; l <= 780; l++) { const e = eps(sp, l); if (e > m) { m = e; b = l; } } return b; }
  /* couleur transmise (sRGB) d'une solution, épaisseur ℓ (cm) : on intègre grossièrement 3 bandes */
  function couleur(sp, C, lc) {
    const ch = [[610, 650], [520, 570], [440, 480]].map(([a, b]) => { let t = 0, n = 0; for (let l = a; l <= b; l += 5) { t += Math.pow(10, -A(sp, C, l, lc)); n++; } return t / n; });
    return ch.map(t => Math.round(255 * Math.pow(Math.max(0, Math.min(1, t)), 1 / 1.6)));
  }
  /* couleur d'une longueur d'onde (pour le faisceau) */
  function rgbLambda(l) {
    let r = 0, g = 0, b = 0;
    if (l < 440) { r = -(l - 440) / 60; b = 1; } else if (l < 490) { g = (l - 440) / 50; b = 1; } else if (l < 510) { g = 1; b = -(l - 510) / 20; }
    else if (l < 580) { r = (l - 510) / 70; g = 1; } else if (l < 645) { r = 1; g = -(l - 645) / 65; } else r = 1;
    const k = l < 420 ? .3 + .7 * (l - 380) / 40 : l > 700 ? .3 + .7 * (780 - l) / 80 : 1;
    return [r, g, b].map(v => Math.round(255 * Math.pow(Math.max(0, v * k), .8)));
  }
  return { ESPECES, eps, A, lmax, couleur, rgbLambda };
})();

  const texte = el => { if (!el) return ""; const c = el.cloneNode(true); c.querySelectorAll("button,input,select,textarea,#efir-bilan-fin").forEach(e => e.remove());
    c.querySelectorAll("h1,h2,h3,h4,p,li,tr,div,br").forEach(e => e.append("\n")); c.querySelectorAll("td,th").forEach(e => e.append(" | "));
    return c.textContent.replace(/[ \t]+/g, " ").replace(/ \| \n/g, "\n").replace(/\n\s*\n+/g, "\n").trim(); };
  const memo = (k, v) => { try { if (v === undefined) return localStorage.getItem("efir-eleve-" + k) || ""; localStorage.setItem("efir-eleve-" + k, v); } catch (e) { return ""; } };
  const fr = x => typeof x === "number" ? (Math.abs(x) >= 1e4 || (x !== 0 && Math.abs(x) < 1e-3) ? x.toExponential(3) : (+x.toPrecision(4)).toString()).replace(".", ",") : String(x);

  function contenu() {
    const h1 = $("h1"), titre = nu(h1 ? h1.innerHTML : document.title) || ID, parts = [];
    let fait = 0, n = 0;
    const et = $("#etapeN"); const m = et && et.textContent.match(/(\d+)\s*\/\s*(\d+)/); if (m) { fait = +m[1]; n = +m[2]; }
    /* paillasse : tableau de mesures + interprétation ; TP sondes : résultats du moteur */
    document.querySelectorAll("table.mesures").forEach(t => { const x = texte(t); if (x) parts.push("Tableau de mesures\n" + x); });
    try { if (typeof ETAPES !== "undefined" && ETAPES) {
      if (ETAPES.finies && ETAPES.finies.size != null && typeof PARTIES !== "undefined") { fait = ETAPES.finies.size; n = PARTIES.length || n; }
      if (ETAPES.res) { const r = Object.entries(ETAPES.res).filter(([k, v]) => v != null && typeof v !== "object").map(([k, v]) => k + " = " + fr(v)); if (r.length) parts.push("Résultats\n" + r.join("\n")); } } } catch (e) {}
    try { if (typeof LAB !== "undefined" && LAB && Array.isArray(LAB.data) && LAB.data.length) parts.push("Mesures enregistrées\n" + LAB.data.slice(0, 80).map(p => Object.entries(p).filter(([k, v]) => typeof v === "number").map(([k, v]) => k + " = " + fr(v)).join(" ; ")).join("\n")); } catch (e) {}
    const interp = texte($("#interp")); if (interp) parts.push("Dernière interprétation\n" + interp);
    const conc = texte($("#etDesc")) || texte($(".etape.active, .etape-courante"));
    return { titre, fait, n, detail: parts.join("\n\n"), conc };
  }

  async function telecharge(nomI, clI, dit, b) {
    const nom = nomI ? nomI.value.trim() : "", cl = clI ? clI.value.trim() : "";
    if (!nom) { dit("Écrivez d'abord votre nom et prénom."); nomI && nomI.focus(); return; }
    if (!cl) { dit("Écrivez d'abord votre classe."); clI && clI.focus(); return; }
    memo("nom", nom); memo("classe", cl);
    const c = contenu(), t1 = Date.now(), av = c.n ? Math.min(1, c.fait / c.n) : 1;
    const o = { v: 1, tp: "act_" + ID, titre: c.titre, eleve: nom, classe: cl, t0, t1, note: null, av: Math.round(av * 100) / 100,
      parties: [[c.titre, c.fait, c.n || c.fait]], erreurs: [], conc: c.conc.slice(0, 1500),
      annexes: [{ titre: "Compte rendu de l'élève", texte: (c.detail + "\n\n" + c.conc).slice(0, 6000) }], j: [] };
    const code = "TPEFIR1:" + btoa(unescape(encodeURIComponent(JSON.stringify(o))));
    const champs = { eleve: nom, classe: cl, tp: c.titre, note: "à corriger", avancement: Math.round(av * 100) + " %", erreurs: "—", reponses_fausses: "",
      temps: Math.max(1, Math.round((t1 - t0) / 60000)) + " min", detail: (c.n ? `Étapes réalisées : ${c.fait} / ${c.n}\n\n` : "") + (c.detail || "—"),
      conclusion: c.conc.slice(0, 3000) || "(conclusion vide)", date: new Date().toLocaleString("fr-FR"), code };
    b.disabled = true; dit("Préparation de votre bilan PDF…");
    try { const f = window.EFIR_BILAN || (window.top && window.top.EFIR_BILAN); if (!f) throw new Error("bilan"); const r = await f(champs); if (!r || !r.ok) throw new Error("pdf");
      dit("✓ Bilan PDF téléchargé : déposez-le dans le devoir Google Classroom."); }
    catch (e) { dit("Le PDF n'a pas pu être créé. Réessayez, ou utilisez « Télécharger mon compte rendu (PDF) » s'il est proposé."); }
    finally { b.disabled = false; }
  }

  const CSS = `#efir-bilan-fin{margin:12px 0;padding:12px 14px;border:1px solid #E1BEE7;border-radius:12px;background:#F8F6FA}
#efir-bilan-fin h3{margin:0 0 4px;font-size:1rem;color:#4A148C}#efir-bilan-fin p{margin:0 0 8px;font-size:.9rem}
#efir-bilan-fin .lig{display:flex;flex-wrap:wrap;gap:8px;margin-bottom:8px}#efir-bilan-fin input{flex:1 1 140px;min-width:0;padding:7px 10px;border:1px solid #D1C4E9;border-radius:8px;font:inherit;background:#fff}
.efir-bilan-btn{background:#4A148C!important;color:#fff!important;border:1px solid #4A148C!important;border-radius:10px;padding:8px 14px;font:inherit;font-weight:700;cursor:pointer}
.efir-bilan-btn:hover:not([disabled]){background:#380E6B!important}.efir-bilan-btn[disabled]{opacity:.6;cursor:wait}
#efir-bilan-note{display:block;margin-top:6px;font-size:.88rem;color:#4A148C}`;
  const st = document.createElement("style"); st.textContent = CSS; document.head.appendChild(st);

  /* paillasse : encadré à l'étape « Conclusion » */
  function paillasse() {
    const desc = $("#etDesc"); if (!desc || !document.body.dataset) return;
    let box = $("#efir-bilan-fin"); const fin = document.body.dataset.etape === "fin";
    if (!fin) { if (box) box.hidden = true; return; }
    if (!box) {
      box = document.createElement("div"); box.id = "efir-bilan-fin";
      box.innerHTML = `<h3>📄 Mon bilan</h3><p>Écrivez votre nom et votre classe, téléchargez le bilan PDF, puis déposez-le dans le devoir Google Classroom.</p>
        <div class="lig"><input id="eb-nom" placeholder="Nom et prénom" autocomplete="name" aria-label="Nom et prénom"><input id="eb-classe" placeholder="Classe" aria-label="Classe"></div>
        <button type="button" class="efir-bilan-btn" id="eb-btn">📄 Télécharger mon bilan</button><span id="efir-bilan-note" role="status" aria-live="polite"></span>`;
      const nav = desc.parentElement.querySelector(".nav"); (nav ? nav.before(box) : desc.after(box));
      const nomI = $("#eb-nom"), clI = $("#eb-classe"); nomI.value = memo("nom"); clI.value = memo("classe");
      $("#eb-btn").addEventListener("click", e => telecharge(nomI, clI, t => $("#efir-bilan-note").textContent = t, e.currentTarget));
    }
    box.hidden = false;
  }
  /* TP sondes : bouton à côté du compte rendu PDF */
  function sondes() {
    const nomI = $("#cr-nom"); if (!nomI || $("#eb-btn-s")) return;
    const cr = [...document.querySelectorAll("button")].find(x => /Télécharger mon compte rendu/.test(x.textContent)); if (!cr) return;
    const b = document.createElement("button"); b.type = "button"; b.id = "eb-btn-s"; b.className = cr.className + " efir-bilan-btn"; b.textContent = "📄 Télécharger mon bilan";
    const note = document.createElement("span"); note.id = "efir-bilan-note"; note.setAttribute("role", "status");
    cr.after(b); b.after(note);
    if (!nomI.value) nomI.value = memo("nom"); const clI = $("#cr-classe"); if (clI && !clI.value) clI.value = memo("classe");
    b.addEventListener("click", () => telecharge($("#cr-nom"), $("#cr-classe"), t => note.textContent = t, b));
  }
  const tour = () => { try { paillasse(); sondes(); } catch (e) {} };
  new MutationObserver(tour).observe(document.documentElement, { subtree: true, childList: true, attributes: true, attributeFilter: ["data-etape"] });
  if (document.readyState === "loading") document.addEventListener("DOMContentLoaded", tour); else tour();
})();
