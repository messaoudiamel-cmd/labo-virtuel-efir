/* ===== Cadrage automatique de la scène 3D (commun aux paillasses réalistes) =====
   - facteur(nom) : distance multipliée pour que, dans la vue « ensemble », tout le matériel ET ses étiquettes
     tiennent dans le cadre, quelle que soit la forme du panneau (portable, grand écran, téléphone) ;
     pour les vues rapprochées, simple correction selon le rapport largeur/hauteur.
   - etiquettes(cam, liste) : chaque étiquette reste entièrement visible (elle se décale vers l'intérieur). */
const CADRAGE = (function () {
  const T = window.THREE, v = () => new T.Vector3();
  function boite(scene, exclure) {
    const B = new T.Box3(), b = new T.Box3(), s = v(); scene.updateMatrixWorld(true);
    scene.traverse(o => {
      if (!o.isMesh || o.isSprite) return; let p = o; while (p) { if (p.visible === false || (exclure && exclure(p))) return; p = p.parent; }
      b.setFromObject(o); if (b.isEmpty()) return; b.getSize(s); if (Math.max(s.x, s.z) > 90 || s.y > 140) return; B.union(b);
    });
    return B;
  }
  function pose(c, V, d) { c.position.set(V.cx + d * Math.sin(V.ph) * Math.sin(V.th), V.cy + d * Math.cos(V.ph), V.cz + d * Math.sin(V.ph) * Math.cos(V.th)); c.lookAt(V.cx, V.cy, V.cz); c.updateMatrixWorld(true); }
  function demi(cam, s) { const t = Math.tan(cam.fov * Math.PI / 360); return [.5 * s.scale.x / (t * cam.aspect), .5 * s.scale.y / t]; }
  /* plus petite distance telle que la boîte et les étiquettes tiennent dans le cadre */
  function distance(cam, V, B, etiq) {
    const pts = []; for (let i = 0; i < 8; i++) pts.push([new T.Vector3(i & 1 ? B.max.x : B.min.x, i & 2 ? B.max.y : B.min.y, i & 4 ? B.max.z : B.min.z), .92, .86]);
    (etiq || []).forEach(s => { if (!s.visible) return; const [hx, hy] = demi(cam, s); pts.push([s.getWorldPosition(v()), .97 - .6 * hx, .95 - .6 * hy]); });
    const c = cam.clone(); let lo = 4, hi = 700;
    for (let k = 0; k < 28; k++) {
      const d = (lo + hi) / 2; pose(c, V, d); let ok = true;
      for (const [p, mx, my] of pts) { const q = p.clone().project(c); if (q.z > 1 || Math.abs(q.x) > Math.max(.2, mx) || Math.abs(q.y) > Math.max(.2, my)) { ok = false; break; } }
      if (ok) hi = d; else lo = d;
    }
    return hi;
  }
  function etiquettes(cam, liste, H) {
    const q = v(), f = 1;
    liste.forEach(s => {
      if (!s.visible) return; s.center.set(.5, .5);
      if (!s.userData.base) s.userData.base = [s.scale.x, s.scale.y]; s.scale.set(s.userData.base[0] * f, s.userData.base[1] * f, 1); s.getWorldPosition(q).project(cam); const [hx, hy] = demi(cam, s);
      const mx = .985 - hx, my = .97 - hy;
      if (q.x > mx) s.center.x = Math.min(1.5, .5 + (q.x - mx) / (2 * hx)); else if (q.x < -mx) s.center.x = Math.max(-.5, .5 + (q.x + mx) / (2 * hx));
      if (q.y > my) s.center.y = Math.min(1.5, .5 + (q.y - my) / (2 * hy)); else if (q.y < -my) s.center.y = Math.max(-.5, .5 + (q.y + my) / (2 * hy));
    });
  }
  /* vue d'ensemble : centre la caméra sur le matériel et règle la distance ; renvoie le facteur de distance */
  function ensemble(cam, scene, V, cible, exclure) {
    const B = boite(scene, exclure); if (B.isEmpty()) return 1; const c = B.getCenter(v());
    const W = Object.assign({}, V, { cx: c.x, cy: c.y - (B.max.y - B.min.y) * .03, cz: c.z });
    Object.assign(cible, { cx: W.cx, cy: W.cy, cz: W.cz }); return distance(cam, W, B, null) / V.d;
  }
  return { boite, distance, etiquettes, ensemble };
})();

  function cyl(a, b, r, ma, seg) { /* cylindre entre deux points */
    const va = new T.Vector3(...a), vb = new T.Vector3(...b), l = va.distanceTo(vb);
    const k = mesh(new T.CylinderGeometry(r, r, l, seg || 20), ma); k.position.copy(va).lerp(vb, .5);
    k.quaternion.setFromUnitVectors(new T.Vector3(0, 1, 0), vb.clone().sub(va).normalize()); return k;
  }
  function boiteArrondie(w, h, d, r, ma) {
    const s = new T.Shape(), x = -w / 2, y = -d / 2;
    s.moveTo(x + r, y); s.lineTo(x + w - r, y); s.quadraticCurveTo(x + w, y, x + w, y + r); s.lineTo(x + w, y + d - r); s.quadraticCurveTo(x + w, y + d, x + w - r, y + d);
    s.lineTo(x + r, y + d); s.quadraticCurveTo(x, y + d, x, y + d - r); s.lineTo(x, y + r); s.quadraticCurveTo(x, y, x + r, y);
    const g = new T.ExtrudeGeometry(s, { depth: h - .4, bevelEnabled: true, bevelThickness: .2, bevelSize: .2, bevelSegments: 3, curveSegments: 6 });
    g.rotateX(-Math.PI / 2); g.translate(0, .2, 0); return mesh(g, ma);
  }

  function seg7(c, txt, x, y, h, coul) { /* afficheur 7 segments dessiné, 4 chiffres + point */
    const S = { "0": "abcdef", "1": "bc", "2": "abged", "3": "abgcd", "4": "fgbc", "5": "afgcd", "6": "afgedc", "7": "abc", "8": "abcdefg", "9": "abcdfg", "-": "g", " ": "" };
    const w = h * .52, e = h * .1, pas = w + h * .2; let cx = x;
    const on = coul, off = "rgba(29,34,21,.07)";
    for (let i = 0; i < txt.length; i++) {
      const ch = txt[i]; if (ch === "." || ch === ",") { c.fillStyle = on; c.fillRect(cx - h * .16, y + h - e, e, e); continue; }
      const segs = S[ch] || ""; const P = {
        a: [cx + e, y, w - 2 * e, e], d: [cx + e, y + h - e, w - 2 * e, e], g: [cx + e, y + h / 2 - e / 2, w - 2 * e, e],
        f: [cx, y + e, e, h / 2 - 1.5 * e], b: [cx + w - e, y + e, e, h / 2 - 1.5 * e], e: [cx, y + h / 2 + e / 2, e, h / 2 - 1.5 * e], c: [cx + w - e, y + h / 2 + e / 2, e, h / 2 - 1.5 * e]
      };
      for (const k in P) { c.fillStyle = segs.includes(k) ? on : off; const [a, b, ww, hh] = P[k]; c.save(); c.translate(a + ww / 2, b + hh / 2); c.transform(1, 0, -.12, 1, 0, 0); c.fillRect(-ww / 2, -hh / 2, ww, hh); c.restore(); }
      cx += pas;
    }
  }

      if (pts.size === 1 && touche(e) && api.surRobinet) { sur = true; api.surRobinet(true); }

    });
    cv.addEventListener("pointermove", e => {
      const p = pts.get(e.pointerId); if (!p) return;
      if (pts.size === 1 && !sur) { const dx = e.clientX - p.x, dy = e.clientY - p.y; if (Math.abs(dx) + Math.abs(dy) > 2) glisse = true; cible.th -= dx * .006; cible.ph = Math.max(.35, Math.min(1.62, cible.ph - dy * .005)); }
      p.x = e.clientX; p.y = e.clientY;

    cv.addEventListener("wheel", e => { e.preventDefault(); cible.d = Math.max(14, Math.min(190, cible.d * (e.deltaY > 0 ? 1.1 : .9))); }, { passive: false });
  }

  let t0 = performance.now(), temps = 0;
  function boucle() {
    requestAnimationFrame(boucle);
    const now = performance.now(), dt = Math.min(.05, (now - t0) / 1000); t0 = now; temps += dt;
    if (api.tick) api.tick(dt);
    const lis = (a, b, k) => a + (b - a) * Math.min(1, dt * k);
    etat.robinet = lis(etat.robinet, etat.robCible, 10); O.cle.rotation.x = etat.robinet * Math.PI / 2;
    etat.agit = lis(etat.agit, etat.agitCible, 2.2); O.boutonVit.rotation.y = -etat.agitCible * 2.4;
    O.voyant.material.emissive.setHex(etat.agitCible > 0 ? 0x33ff66 : 0x000000); O.voyant.material.color.setHex(etat.agitCible > 0 ? 0x66ff88 : 0x3a3a3a);
    { /* déplacement de la sonde : monter, se déplacer, descendre */
      const P = O.cellule.position, L = etat.celLieu && O.tampons ? O.tampons[etat.celLieu] : null;
      const tx = L ? L.x : CEL.x, tz = L ? L.z : CEL.z, ty = L ? .55 : Y_PLAT + 1.25 + (1 - etat.celCible) * 12;
      const loin = Math.hypot(P.x - tx, P.z - tz) > .3, k = Math.min(1, dt * 3.2);
      if (loin && P.y < 24) P.y += (25 - P.y) * k; else if (loin) { P.x += (tx - P.x) * k; P.z += (tz - P.z) * k; if (Math.hypot(P.x - tx, P.z - tz) < .35) { P.x = tx; P.z = tz; } }
      else { P.x = tx; P.z = tz; P.y += (ty - P.y) * k; }

    const nomI = $("#cr-nom"); if (!nomI || $("#eb-btn-s")) return;
    const cr = [...document.querySelectorAll("button")].find(x => /Télécharger mon compte rendu/.test(x.textContent)); if (!cr) return;
    const b = document.createElement("button"); b.type = "button"; b.id = "eb-btn-s"; b.className = cr.className + " efir-bilan-btn"; b.textContent = "📄 Télécharger mon bilan";
    const note = document.createElement("span"); note.id = "efir-bilan-note"; note.setAttribute("role", "status");
    cr.after(b); b.after(note);
    if (!nomI.value) nomI.value = memo("nom"); const clI = $("#cr-classe"); if (clI && !clI.value) clI.value = memo("classe");
    b.addEventListener("click", () => telecharge($("#cr-nom"), $("#cr-classe"), t => note.textContent = t, b));
  }
  const tour = () => { try { paillasse(); sondes(); } catch (e) {} };
  new MutationObserver(tour).observe(document.documentElement, { subtree: true, childList: true, attributes: true, attributeFilter: ["data-etape"] });
  if (document.readyState === "loading") document.addEventListener("DOMContentLoaded", tour); else tour();
})();
