const sig17 = x => 1 / (1 + Math.exp(-x));
function specVal17(key, l) {
  const L = LIGHTS17[key]; if (L.line) return Math.exp(-(((l - L.line) / 3) ** 2));
  const lamp = .35 + .65 * (l - 380) / 400; let t = 0;
  for (const k of L.b) { const [a, b] = BANDS17[k]; t += sig17((l - a + (k === 'B' ? 25 : 0)) / 7) * sig17((b - l + (k === 'R' ? 60 : 0)) / 7); } t = Math.min(1, t);
  return lamp * (key === 'blanche' ? 1 : .05 + .9 * t);
}
function wl2rgb17(l) {
  let r = 0, g = 0, b = 0;
  if (l < 440) { r = -(l - 440) / 60; b = 1; } else if (l < 490) { g = (l - 440) / 50; b = 1; } else if (l < 510) { g = 1; b = -(l - 510) / 20; } else if (l < 580) { r = (l - 510) / 70; g = 1; } else if (l < 645) { r = 1; g = -(l - 645) / 65; } else r = 1;
  const f = l < 420 ? .3 + .7 * (l - 380) / 40 : l > 700 ? .3 + .7 * (780 - l) / 80 : 1;
  return `rgb(${Math.round(255 * r * f)},${Math.round(255 * g * f)},${Math.round(255 * b * f)})`;
}

      const dE = ei - ef, lam = lam30(dE); s.trans = s.trans.filter(t => !(t.i === a.i && t.f === a.f)); s.trans.push({ i: a.i, f: a.f, dE, lam }); this.drawDia(); this.drawData();
      if (a.i === 6 && a.f === 3) step('t63'); if (a.i === 3 && a.f === 1) step('germ');
      const dom = lam < 380 ? 'ultraviolet' : lam > 780 ? 'infrarouge' : 'visible';
      const msg = `Transition ${a.i} → ${a.f} : ℰphoton = ${f30(ei, 2)} − (${f30(ef, 2)}) = ${f30(dE, 2)} eV = ${sci30(dE * EV30)} J → λ = h × c / ℰphoton = ${f30(lam, 0)} nm (${dom}).`;
      observe(msg); say(msg); return;
    }
    if (a.t === 'clear') { s.reads = []; s.trans = []; this.place(); this.drawData(); say('Relevés effacés.'); return; }
  },
