/* ===== Cadrage automatique de la scène 3D (commun aux paillasses réalistes) =====
   - facteur(nom) : distance multipliée pour que, dans la vue « ensemble », tout le matériel ET ses étiquettes
     tiennent dans le cadre, quelle que soit la forme du panneau (portable, grand écran, téléphone) ;
     pour les vues rapprochées, simple correction selon le rapport largeur/hauteur.
   - etiquettes(cam, liste) : chaque étiquette reste entièrement visible (elle se décale vers l'intérieur). */
const CADRAGE = (function () {
  const T = window.THREE, v = () => new T.Vector3();
  function boite(scene, exclure) {
    const B = new T.Box3(), b = new T.Box3(), s = v(); scene.updateMatrixWorld(true);
    scene.traverse(o => {
      if (!o.isMesh || o.isSprite) return; let p = o; while (p) { if (p.visible === false || (exclure && exclure(p))) return; p = p.parent; }
      b.setFromObject(o); if (b.isEmpty()) return; b.getSize(s); if (Math.max(s.x, s.z) > 90 || s.y > 140) return; B.union(b);
    });
    return B;
  }
  function pose(c, V, d) { c.position.set(V.cx + d * Math.sin(V.ph) * Math.sin(V.th), V.cy + d * Math.cos(V.ph), V.cz + d * Math.sin(V.ph) * Math.cos(V.th)); c.lookAt(V.cx, V.cy, V.cz); c.updateMatrixWorld(true); }
  function demi(cam, s) { const t = Math.tan(cam.fov * Math.PI / 360); return [.5 * s.scale.x / (t * cam.aspect), .5 * s.scale.y / t]; }
  /* plus petite distance telle que la boîte et les étiquettes tiennent dans le cadre */
  function distance(cam, V, B, etiq) {
    const pts = []; for (let i = 0; i < 8; i++) pts.push([new T.Vector3(i & 1 ? B.max.x : B.min.x, i & 2 ? B.max.y : B.min.y, i & 4 ? B.max.z : B.min.z), .92, .86]);
    (etiq || []).forEach(s => { if (!s.visible) return; const [hx, hy] = demi(cam, s); pts.push([s.getWorldPosition(v()), .97 - .6 * hx, .95 - .6 * hy]); });
    const c = cam.clone(); let lo = 4, hi = 700;
    for (let k = 0; k < 28; k++) {
      const d = (lo + hi) / 2; pose(c, V, d); let ok = true;
      for (const [p, mx, my] of pts) { const q = p.clone().project(c); if (q.z > 1 || Math.abs(q.x) > Math.max(.2, mx) || Math.abs(q.y) > Math.max(.2, my)) { ok = false; break; } }
      if (ok) hi = d; else lo = d;
    }
    return hi;
  }
  function etiquettes(cam, liste, H) {
    const q = v(), f = 1;
    liste.forEach(s => {
      if (!s.visible) return; s.center.set(.5, .5);
      if (!s.userData.base) s.userData.base = [s.scale.x, s.scale.y]; s.scale.set(s.userData.base[0] * f, s.userData.base[1] * f, 1); s.getWorldPosition(q).project(cam); const [hx, hy] = demi(cam, s);
      const mx = .985 - hx, my = .97 - hy;
      if (q.x > mx) s.center.x = Math.min(1.5, .5 + (q.x - mx) / (2 * hx)); else if (q.x < -mx) s.center.x = Math.max(-.5, .5 + (q.x + mx) / (2 * hx));
      if (q.y > my) s.center.y = Math.min(1.5, .5 + (q.y - my) / (2 * hy)); else if (q.y < -my) s.center.y = Math.max(-.5, .5 + (q.y + my) / (2 * hy));
    });
  }
  /* vue d'ensemble : centre la caméra sur le matériel et règle la distance ; renvoie le facteur de distance */
  function ensemble(cam, scene, V, cible, exclure) {
    const B = boite(scene, exclure); if (B.isEmpty()) return 1; const c = B.getCenter(v());
    const W = Object.assign({}, V, { cx: c.x, cy: c.y - (B.max.y - B.min.y) * .03, cz: c.z });
    Object.assign(cible, { cx: W.cx, cy: W.cy, cz: W.cz }); return distance(cam, W, B, null) / V.d;
  }
  return { boite, distance, etiquettes, ensemble };
})();
