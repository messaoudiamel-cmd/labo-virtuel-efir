const GG14 = 9.81;
// données brutes [{t, x, y}] → ajoute v (dérivée centrée, comme un logiciel de pointage) et les énergies
function energize(raw, m) {
  return raw.map((p, i) => {
    const a = raw[Math.max(0, i - 1)], b = raw[Math.min(raw.length - 1, i + 1)], dt = b.t - a.t;
    const v = Math.hypot(b.x - a.x, b.y - a.y) / dt;
    const Ec = .5 * m * v * v, Ep = m * GG14 * p.y; return { ...p, v, Ec, Ep, Em: Ec + Ep };
  }).slice(1, -1);
}

}

function simulate14(th0deg) { // RK4, θ'' = −(g/L) sin θ ; départ sans vitesse en θ0 (côté x > 0)
  let th = th0deg * Math.PI / 180, w = 0; const h = .002, out = []; let t = 0, next = 0;
  const f = (th, w) => [w, -GG14 / LP * Math.sin(th)];
  while (t <= TREC + 1e-9) {
    if (t >= next - 1e-9) { const nz = (Math.sin(t * 97.3) + Math.sin(t * 41.1)) * .0004; out.push({ t: +next.toFixed(2), x: +(LP * Math.sin(th) + nz).toFixed(4), y: +(LP * (1 - Math.cos(th)) + nz * .5).toFixed(4), th }); next += DT14; }
    const k1 = f(th, w), k2 = f(th + h / 2 * k1[0], w + h / 2 * k1[1]), k3 = f(th + h / 2 * k2[0], w + h / 2 * k2[1]), k4 = f(th + h * k3[0], w + h * k3[1]);
    th += h / 6 * (k1[0] + 2 * k2[0] + 2 * k3[0] + k4[0]); w += h / 6 * (k1[1] + 2 * k2[1] + 2 * k3[1] + k4[1]); t += h;
  }
  return out;
}
