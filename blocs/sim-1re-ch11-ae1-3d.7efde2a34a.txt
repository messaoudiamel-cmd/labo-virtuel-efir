function makeCup(k) {

  const stem = new THREE.Mesh(new THREE.CylinderGeometry(.003, .003, .02, 8), rub); stem.position.y = -r * .6 - .01; g.add(stem);
  const hook = new THREE.Mesh(new THREE.TorusGeometry(.004, .0012, 6, 16), new THREE.MeshStandardMaterial({ color: 0x9e9e9e, metalness: .8 })); hook.position.y = -r * .6 - .024; g.add(hook);

  const mass = new THREE.Group(); mass.position.y = -r * .6 - .068; g.add(mass); g.mass = mass;
  const l = label((k === 'small' ? 'Petite' : 'Grande') + `\nd = ${String(c.d).replace('.', ',')} cm`, { size: 22, scale: .0006 }); l.position.set(c.lx, -.03, .03); g.add(l);
  g.home = new THREE.Vector3(c.x, YPL, 0); g.table = new THREE.Vector3(c.x * 4 + .35, .005 + r * .6 + .08, .25);
  return g;
}

}
