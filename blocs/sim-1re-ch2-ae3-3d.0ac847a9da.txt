function grenaille() {
  const mat = new THREE.MeshStandardMaterial({ color: 0x5b5f66, metalness: .7, roughness: .45 });
  const n = 26, im = new THREE.InstancedMesh(new THREE.IcosahedronGeometry(.0085, 0), mat, n), m = new THREE.Matrix4(), q = new THREE.Quaternion(), e = new THREE.Euler(), v = new THREE.Vector3(), sc = new THREE.Vector3();
  for (let i = 0; i < n; i++) { const a = rnd(0, 6.28), r = Math.sqrt(Math.random()) * .038; e.set(rnd(0, 3), rnd(0, 3), rnd(0, 3)); q.setFromEuler(e); v.set(Math.cos(a) * r, rnd(0, .012) + r * r * 3, Math.sin(a) * r); const s = rnd(.8, 1.3); sc.set(s, s * .8, s); m.compose(v, q, sc); im.setMatrixAt(i, m); }
  const g = new THREE.Group(); g.add(im); g.userData.mat = mat; return g;
}

function filCuivre() {
  const g = new THREE.Group(), pts = [];
  for (let k = 0; k <= 60; k++) { const a = k * .45; pts.push(new THREE.Vector3(Math.cos(a) * .045, .01 + k * .0035, Math.sin(a) * .045)); }
  pts.push(new THREE.Vector3(0, .3, 0), new THREE.Vector3(.01, .55, 0));
  const mat = new THREE.MeshStandardMaterial({ color: 0xc9773f, metalness: .75, roughness: .35 });
  g.add(new THREE.Mesh(new THREE.TubeGeometry(new THREE.CatmullRomCurve3(pts), 160, .0045, 6), mat));
  const am = new THREE.MeshStandardMaterial({ color: 0xc9ccd2, metalness: .6, roughness: .5 }), N = 90, im = new THREE.InstancedMesh(new THREE.BoxGeometry(.004, .022, .004), am, N), M = new THREE.Object3D(); g.add(im);
  const pousse = p => { for (let k = 0; k < N; k++) { const q = pts[(k * 7) % 58], a = k * 1.9; M.position.set(q.x + Math.cos(a) * .008, q.y, q.z + Math.sin(a) * .008); M.rotation.set(Math.cos(a), a, Math.sin(a));
    M.scale.setScalar(p > k / N * .85 ? Math.min(1, p * 1.5) : .001); M.updateMatrix(); im.setMatrixAt(k, M.matrix); } im.instanceMatrix.needsUpdate = true;
    mat.color.setRGB(lerp(.79, .6, p), lerp(.47, .42, p), lerp(.25, .3, p)); };
  pousse(0); g.traverse(o => { if (o.isMesh) o.castShadow = true; });
  return { g, pousse };
}
/* vue microscopique : un atome de cuivre cède 2 électrons à deux ions argent */
function simArgent(sc) {
  const S = .5, geoA = new THREE.SphereGeometry(.24, 24, 16), geoI = new THREE.SphereGeometry(.2, 24, 16), geoE = new THREE.SphereGeometry(.06, 12, 8);
  const mat = c => new THREE.MeshStandardMaterial({ color: c, metalness: .3, roughness: .45 }), matIon = c => new THREE.MeshStandardMaterial({ color: c, emissive: c, emissiveIntensity: .35, roughness: .4 });
  const atoms = [], ions = [], box = { x: 3.6, y0: 1.1, y1: 3.9, z: 1.8 }, haut = new Map(); let events = [], tEv = .6;
  for (let i = -7; i <= 7; i++) for (let k = -3; k <= 3; k++) for (let j = 0; j < 2; j++) { const a = new THREE.Mesh(geoA, mat(0xc9773f)); a.position.set(i * S, -j * S, k * S); sc.add(a); if (!j) { atoms.push(a); a.userData = { i, k }; } haut.set(i + ',' + k, 0); }
  const lab = label('Cuivre métallique Cu(s)', { size: 40, scale: .007, bg: 'rgba(255,255,255,.9)' }); lab.position.set(-3.2, -.9, 2.2); sc.add(lab);
  const ajoute = () => { const m = new THREE.Mesh(geoI, matIon(0xe0e0e0)); m.position.set(rnd(-box.x, box.x), rnd(box.y0, box.y1), rnd(-box.z, box.z)); m.add(ionLabel('Ag⁺', '#fff')); m.userData = { v: new THREE.Vector3(rnd(-.3, .3), rnd(-.2, .2), rnd(-.3, .3)) }; sc.add(m); ions.push(m); };
  for (let n = 0; n < 12; n++) ajoute();
  const brown = (m, dt) => { const v = m.userData.v; v.add(new THREE.Vector3(rnd(-.6, .6), rnd(-.5, .5), rnd(-.6, .6)).multiplyScalar(dt)); v.clampLength(0, .45); m.position.addScaledVector(v, dt);
    if (Math.abs(m.position.x) > box.x) v.x *= -1; if (Math.abs(m.position.z) > box.z) v.z *= -1; if (m.position.y < box.y0 || m.position.y > box.y1 + 1.5) v.y *= -1; };
  const arc = (a, b, p, hgt) => { const m = a.clone().lerp(b, p); m.y += Math.sin(p * Math.PI) * hgt; return m; };
  function nouveau() {
    const libres = atoms.filter(a => !a.userData.busy && Math.abs(a.userData.i) < 6 && Math.abs(a.userData.k) < 3); const cand = ions.filter(m => !m.userData.busy);
    if (!libres.length || cand.length < 2) return; const A = libres[Math.floor(Math.random() * libres.length)]; A.userData.busy = true;
    cand.sort((a, b) => a.position.distanceTo(A.position) - b.position.distanceTo(A.position)); const I = cand.slice(0, 2); I.forEach(m => m.userData.busy = true);
    const sites = [-1, 1].map(d => { const ni = THREE.MathUtils.clamp(A.userData.i + d, -7, 7), key = ni + ',' + A.userData.k, hh = haut.get(key) + 1; haut.set(key, hh); return new THREE.Vector3(ni * S, hh * S * .9, A.userData.k * S); });
    events.push({ t: 0, A, I, p0: I.map(m => m.position.clone()), sites });
  }
  function pas(ev, dt) {
    ev.t += dt; const t = ev.t, A = ev.A;
    if (t < 1.5) ev.I.forEach((m, k) => m.position.copy(ev.p0[k].clone().lerp(ev.sites[k].clone().add(new THREE.Vector3(0, .9, 0)), ease(t / 1.5))));
    else if (t < 2.8) { if (!ev.e) ev.e = [0, 1].map(() => { const e = new THREE.Mesh(geoE, new THREE.MeshBasicMaterial({ color: 0xffeb3b })); sc.add(e); return e; });
      const p = (t - 1.5) / 1.3; ev.e.forEach((e, k) => e.position.copy(arc(A.position, ev.I[k].position, Math.max(0, Math.min(1, p)), .7))); }
    else if (!ev.fait) { ev.fait = true; ev.e.forEach(e => sc.remove(e)); const c = $('#microCount'); if (c) c.textContent = String((+c.textContent || 0) + 2);
      A.material = matIon(0x29b6f6); A.add(ionLabel('Cu²⁺')); ev.I.forEach(m => { m.remove(m.children[0]); m.material = mat(0xd5d8de); m.geometry = geoA; ions.splice(ions.indexOf(m), 1); }); ev.I0 = ev.I.map(m => m.position.clone()); }
    else if (t < 4) { ev.I.forEach((m, k) => m.position.copy(ev.I0[k].clone().lerp(ev.sites[k], ease((t - 2.8) / 1.2)))); A.position.y += dt * .6; }
    else { atoms.splice(atoms.indexOf(A), 1); A.userData = { v: new THREE.Vector3(rnd(-.3, .3), .25, rnd(-.3, .3)), libre: true }; ev.fin = true; ajoute(); ajoute(); }
  }
  const partis = [];
  return dt => {
    ions.forEach(m => { if (!m.userData.busy) brown(m, dt); });
    sc.children.forEach(o => { if (o.userData && o.userData.libre) { if (!partis.includes(o)) partis.push(o); } });
    partis.forEach(o => brown(o, dt));
    tEv -= dt; if (tEv < 0 && events.length < 2) { nouveau(); tEv = 2.6; }
    events.forEach(ev => pas(ev, dt)); events = events.filter(e => !e.fin);
  };
}
