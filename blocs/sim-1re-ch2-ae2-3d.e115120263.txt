function checkCuve() { if (tubes2.some(t => t && t.m === 'X' && t.popped && t.naoh)) step('cuve'); }

function select2(k, silent) {
  sel2 = k; tubes2.forEach((T, j) => T && highlight(T, j === k)); AE2.refreshPanel();
  if (!silent && tubes2[k]) say(`Tube ${k + 1} sélectionné : ${M2[tubes2[k].m].nom} + acide chlorhydrique.`);
}
function prepare2() {
  if (!chosen2) return say('Choisis d\'abord un métal.');
  if (tubes2.some(t => t && t.m === chosen2)) return say('Ce métal est déjà dans un tube.');
  const slot = tubes2.findIndex(t => !t); if (slot < 0) return say('Le portoir est plein : vide les tubes dans le bidon pour recommencer.');
  act({ t: 'prep', slot, metal: chosen2 });
}
