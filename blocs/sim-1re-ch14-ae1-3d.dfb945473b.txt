const GG14 = 9.81;
// données brutes [{t, x, y}] → ajoute v (dérivée centrée, comme un logiciel de pointage) et les énergies
function energize(raw, m) {
  return raw.map((p, i) => {
    const a = raw[Math.max(0, i - 1)], b = raw[Math.min(raw.length - 1, i + 1)], dt = b.t - a.t;
    const v = Math.hypot(b.x - a.x, b.y - a.y) / dt;
    const Ec = .5 * m * v * v, Ep = m * GG14 * p.y; return { ...p, v, Ec, Ep, Em: Ec + Ep };
  }).slice(1, -1);
}
