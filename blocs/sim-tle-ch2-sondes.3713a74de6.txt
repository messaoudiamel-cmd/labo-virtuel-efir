const OUTILS = {
  trie(d, cle) { const m = new Map(); d.forEach(p => { if (Number.isFinite(p[cle])) m.set(+p.V.toFixed(3), p[cle]); }); return [...m.entries()].sort((a, b) => a[0] - b[0]); },
  pchip(P) {
    const n = P.length, x = P.map(p => p[0]), y = P.map(p => p[1]), h = [], dl = [], mm = new Array(n).fill(0);
    for (let i = 0; i < n - 1; i++) { h[i] = x[i + 1] - x[i]; dl[i] = (y[i + 1] - y[i]) / h[i]; }
    if (n < 2) return null;
    mm[0] = dl[0]; mm[n - 1] = dl[n - 2];
    for (let i = 1; i < n - 1; i++) { if (dl[i - 1] * dl[i] <= 0) mm[i] = 0; else { const w1 = 2 * h[i] + h[i - 1], w2 = h[i] + 2 * h[i - 1]; mm[i] = (w1 + w2) / (w1 / dl[i - 1] + w2 / dl[i]); } }
    const seg = v => { let i = 0; while (i < n - 2 && v > x[i + 1]) i++; return i; };
    return {
      xmin: x[0], xmax: x[n - 1],
      f(v) { const i = seg(v), t = (v - x[i]) / h[i], t2 = t * t, t3 = t2 * t; return (2 * t3 - 3 * t2 + 1) * y[i] + (t3 - 2 * t2 + t) * h[i] * mm[i] + (-2 * t3 + 3 * t2) * y[i + 1] + (t3 - t2) * h[i] * mm[i + 1]; },
      d(v) { const i = seg(v), t = (v - x[i]) / h[i], t2 = t * t; return ((6 * t2 - 6 * t) * y[i] + (3 * t2 - 4 * t + 1) * h[i] * mm[i] + (-6 * t2 + 6 * t) * y[i + 1] + (3 * t2 - 2 * t) * h[i] * mm[i + 1]) / h[i]; }
    };
  },
  derivee(P) { const r = []; for (let i = 0; i < P.length - 1; i++) { const dx = P[i + 1][0] - P[i][0]; if (dx > 0) r.push([(P[i][0] + P[i + 1][0]) / 2, Math.abs((P[i + 1][1] - P[i][1]) / dx)]); } return r; },
  veDerivee(P) {
    const D = this.derivee(P); if (D.length < 3) return null; let k = 0; D.forEach((p, i) => { if (p[1] > D[k][1]) k = i; });
    if (k === 0 || k === D.length - 1) return { V: D[k][0], D };
    const [a, b, c] = [D[k - 1], D[k], D[k + 1]], den = (a[0] - b[0]) * (a[0] - c[0]) * (b[0] - c[0]);
    if (!den) return { V: b[0], D };
    const A = (c[0] * (b[1] - a[1]) + b[0] * (a[1] - c[1]) + a[0] * (c[1] - b[1])) / den, B = (c[0] * c[0] * (a[1] - b[1]) + b[0] * b[0] * (c[1] - a[1]) + a[0] * a[0] * (b[1] - c[1])) / den;
    const V = A < 0 ? -B / (2 * A) : b[0]; return { V: Math.min(c[0], Math.max(a[0], V)), D };
  },
  tangentes(P, V1) {
    const S = this.pchip(P), E = this.veDerivee(P); if (!S || !E || P.length < 6) return null;
    V1 = Math.max(S.xmin + .05, Math.min(E.V - .3, V1)); const s = S.d(V1), b1 = S.f(V1) - s * V1;
    let V2 = null; for (let v = E.V + .05; v <= S.xmax; v += .01) { if (S.d(v) <= s) { V2 = v; break; } }
    if (V2 == null) return { erreur: "Pas assez de points après l'équivalence pour tracer la tangente parallèle." };
    const b2 = S.f(V2) - s * V2, bm = (b1 + b2) / 2, g = v => S.f(v) - (s * v + bm);
    let lo = V1, hi = V2; if (g(lo) * g(hi) > 0) return { erreur: "Déplacez la première tangente." };
    for (let i = 0; i < 60; i++) { const mi = (lo + hi) / 2; if (g(lo) * g(mi) <= 0) hi = mi; else lo = mi; }
    const VE = (lo + hi) / 2; return { V1, V2, s, b1, b2, bm, VE, pHE: S.f(VE) };
  },
  ajuste(P, a, b) {
    const Q = P.filter(p => p[0] >= Math.min(a, b) - 1e-9 && p[0] <= Math.max(a, b) + 1e-9); if (Q.length < 2) return null;
    const n = Q.length, sx = Q.reduce((s, p) => s + p[0], 0), sy = Q.reduce((s, p) => s + p[1], 0), sxx = Q.reduce((s, p) => s + p[0] * p[0], 0), sxy = Q.reduce((s, p) => s + p[0] * p[1], 0), syy = Q.reduce((s, p) => s + p[1] * p[1], 0);
    const den = n * sxx - sx * sx; if (!den) return null; const m = (n * sxy - sx * sy) / den, k = (sy - m * sx) / n;
    const r = (n * sxy - sx * sy) / Math.sqrt(den * (n * syy - sy * sy) || 1); return { m, k, r, n, a: Math.min(a, b), b: Math.max(a, b) };
  },
  ajusteProp(P) {
    if (P.length < 2) return null; const sxx = P.reduce((s, p) => s + p[0] * p[0], 0), sxy = P.reduce((s, p) => s + p[0] * p[1], 0), m = sxy / sxx;
    const my = P.reduce((s, p) => s + p[1], 0) / P.length, sst = P.reduce((s, p) => s + (p[1] - my) ** 2, 0), sse = P.reduce((s, p) => s + (p[1] - m * p[0]) ** 2, 0);
    return { m, k: 0, r: Math.sqrt(Math.max(0, 1 - sse / sst)), n: P.length, a: Math.min(...P.map(p => p[0])), b: Math.max(...P.map(p => p[0])), prop: true };
  },
  pas(e) { const p = Math.pow(10, Math.floor(Math.log10(e))), r = e / p; return (r < 1.5 ? 1 : r < 3.5 ? 2 : r < 7.5 ? 5 : 10) * p; }
};
