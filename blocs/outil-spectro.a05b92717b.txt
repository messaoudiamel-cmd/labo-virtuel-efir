        <details class="aide"><summary>❓ Aide : lire un spectre</summary><ul>
          <li>Le <b>blanc</b> se fait avec une cuve remplie du solvant seul : l'appareil retire ainsi l'absorption de la cuve et du solvant.</li>
          <li><b>λmax</b> est la longueur d'onde où l'absorbance est maximale. On y fait les mesures d'étalonnage : c'est là que la mesure est la plus sensible.</li>
          <li>La solution absorbe surtout autour de λmax : sa <b>couleur perçue</b> est proche de la couleur complémentaire de la couleur absorbée (cercle chromatique).</li>
          <li>Transmittance et absorbance sont liées par <code>A = −log(T)</code> (avec T entre 0 et 1).</li>
          <li>Import CSV : deux colonnes, la longueur d'onde en nm puis l'absorbance (séparateur ; ou tabulation, virgule décimale acceptée).</li>
        </ul></details>

  const d = Math.floor(a / pas + 1e-9) * pas, f = Math.ceil(b / pas - 1e-9) * pas, t = []; for (let v = d; v <= f + pas / 2; v += pas) t.push(Math.abs(v) < pas * 1e-9 ? 0 : v); return { d, f, pas, t }; }

function trace(cv, o) {
  const dpr = window.devicePixelRatio || 1, W = cv.clientWidth || 640, H = cv.clientHeight || 380; cv.width = Math.round(W * dpr); cv.height = Math.round(H * dpr);
  const g = cv.getContext("2d"); g.setTransform(dpr, 0, 0, dpr, 0, 0); g.clearRect(0, 0, W, H); g.fillStyle = "#fff"; g.fillRect(0, 0, W, H);
  g.font = "13px Atkinson Hyperlegible, sans-serif";
  const SS = o.series.filter(s => s.x && s.x.length), xs = [], ys = [];
  SS.forEach(s => s.y.forEach((y, i) => { if (isFinite(y) && isFinite(s.x[i])) { xs.push(s.x[i]); if (!s.horsEchelle) ys.push(y); } }));
  const fixeX = isFinite(o.x0) && isFinite(o.x1);
  if (!xs.length && !(fixeX && isFinite(o.y0) && isFinite(o.y1))) { g.fillStyle = "#5E4D6B"; g.textAlign = "center"; g.fillText(o.vide || "Pas encore de mesures à tracer", W / 2, H / 2); cv._geo = null; return null; }
  let x0 = fixeX ? o.x0 : Math.min(...xs), x1 = fixeX ? o.x1 : Math.max(...xs);
  let y0 = isFinite(o.y0) ? o.y0 : ys.length ? Math.min(...ys) : 0, y1 = isFinite(o.y1) ? o.y1 : ys.length ? Math.max(...ys) : 1;
  if (o.zero) { x0 = Math.min(0, x0); x1 = Math.max(0, x1); y0 = Math.min(0, y0); y1 = Math.max(0, y1); }
  if (o.ymin0 && y0 > 0) y0 = 0;
  const gxT = graduations(x0, x1, W < 500 ? 5 : 9), gyT = graduations(y0, y1, 6);
  if (o.serre) { gxT.d = x0; gxT.f = x1; gxT.t = gxT.t.filter(v => v >= x0 - 1e-9 && v <= x1 + 1e-9); }
  const leg = SS.filter(s => s.nom && !s.sansLegende), bh = o.bande ? 12 : 0;

  const sx = v => m.g + (v - gxT.d) / (gxT.f - gxT.d) * pw, sy = v => m.h + ph - (v - gyT.d) / (gyT.f - gyT.d) * ph;
  const geo = { sx, sy, m, pw, ph, W, H, series: SS, xr: [gxT.d, gxT.f], yr: [gyT.d, gyT.f], inv: (px, py) => [gxT.d + (px - m.g) / pw * (gxT.f - gxT.d), gyT.d + (m.h + ph - py) / ph * (gyT.f - gyT.d)] };
  g.strokeStyle = "#F3E5F5"; g.lineWidth = 1; g.fillStyle = "#5E4D6B";
  g.textAlign = "center"; gxT.t.forEach(v => { const p = sx(v); g.beginPath(); g.moveTo(p, m.h); g.lineTo(p, m.h + ph); g.stroke(); g.fillText(fmt(v, 3), p, m.h + ph + bh + 16); });

  if (o.bande) for (let p = 0; p < pw; p++) { const l = gxT.d + p / pw * (gxT.f - gxT.d); g.fillStyle = lcoul(l); g.fillRect(m.g + p, m.h + ph + 1, 1.5, bh - 2); }

  if (gxT.d < 0 && gxT.f > 0) { g.strokeStyle = "#B39DDB"; g.beginPath(); g.moveTo(sx(0), m.h); g.lineTo(sx(0), m.h + ph); g.stroke(); }

  g.save(); g.translate(14, m.h + ph / 2); g.rotate(-Math.PI / 2); g.fillText(o.yl || "", 0, 0); g.restore();

    if (s.ligne) { g.lineWidth = s.larg || 2; g.beginPath(); let d = false; s.y.forEach((y, i) => { if (!isFinite(y) || !isFinite(s.x[i])) { d = false; return; } d ? g.lineTo(sx(s.x[i]), sy(y)) : g.moveTo(sx(s.x[i]), sy(y)); d = true; }); g.stroke(); }
    g.setLineDash([]);
    if (s.pts) { g.lineWidth = 1.6; s.y.forEach((y, i) => { if (!isFinite(y) || !isFinite(s.x[i])) return; const a = sx(s.x[i]), b = sy(y), r = 4.5; g.beginPath(); g.moveTo(a - r, b); g.lineTo(a + r, b); g.moveTo(a, b - r); g.lineTo(a, b + r); g.stroke(); }); }

  g.restore();

  cv._geo = geo; return geo;
}

function survol(cv, texte) {
  const b = cv.parentNode.querySelector(".bulle");

}

function resout(A, b) { const n = b.length, M = A.map((r, i) => r.concat([b[i]]));
  for (let c = 0; c < n; c++) { let p = c; for (let r = c + 1; r < n; r++) if (Math.abs(M[r][c]) > Math.abs(M[p][c])) p = r; [M[c], M[p]] = [M[p], M[c]]; if (Math.abs(M[c][c]) < 1e-300) return null;

function ajuste(f, noms, p0, X, Y) {
  let p = p0.slice(), lam = 1e-3;
  const res = q => { const r = f(q); let s = 0; for (let i = 0; i < Y.length; i++) { const d = Y[i] - r[i]; s += d * d; } return { r, s }; };
  let cur = res(p); if (!isFinite(cur.s)) return null;

    const JTJ = noms.map((_, a) => noms.map((_, b) => { let s = 0; for (let i = 0; i < Y.length; i++) s += J[a][i] * J[b][i]; return s; }));
    const g = noms.map((_, a) => { let s = 0; for (let i = 0; i < Y.length; i++) s += J[a][i] * (Y[i] - cur.r[i]); return s; });
    let ok = false;
    for (let t = 0; t < 12; t++) { const A = JTJ.map((r, a) => r.map((v, b) => a === b ? v * (1 + lam) + 1e-30 : v)); const d = resout(A, g);
      if (d) { const q = p.map((v, k) => v + d[k]), n = res(q); if (isFinite(n.s) && n.s <= cur.s) { const gain = cur.s - n.s; p = q; cur = n; lam = Math.max(lam / 4, 1e-12); ok = true; if (gain <= 1e-14 * (cur.s + 1e-300)) it = 1e9; break; } }

    if (!ok) break;
  }
  return { p, s: cur.s };
}

function uniteParam(mo, k) { const ux = uniteDe(mo.x), uy = uniteDe(mo.y), q = (a, b) => a && b ? a + "/" + b : b ? "1/" + b : a;
